-- СОЗДАНИЕ ЭФФЕКТОВ (ОСНОВНЫЕ)
-- ============================================
local function CreateEffects()
    for i, v in pairs(Lighting:GetChildren()) do
        if v:IsA("BloomEffect") or v:IsA("BlurEffect") or v:IsA("ColorCorrectionEffect") or 
           v:IsA("SunRaysEffect") or v:IsA("Sky") or v:IsA("Atmosphere") then
            v:Destroy()
        end
    end

    local Bloom = Instance.new("BloomEffect")
    local Blur = Instance.new("BlurEffect")
    local ColorCor = Instance.new("ColorCorrectionEffect")
    local SunRays = Instance.new("SunRaysEffect")
    local Sky = Instance.new("Sky")
    local Atm = Instance.new("Atmosphere")

    Bloom.Parent = Lighting
    Blur.Parent = Lighting
    ColorCor.Parent = Lighting
    SunRays.Parent = Lighting
    Sky.Parent = Lighting
    Atm.Parent = Lighting

    Bloom.Intensity = 0.3
    Bloom.Size = 10
    Bloom.Threshold = 0.8

    Blur.Size = 5

    ColorCor.Brightness = 0.1
    ColorCor.Contrast = 0.5
    ColorCor.Saturation = -0.3
    ColorCor.TintColor = Color3.fromRGB(255, 235, 203)

    SunRays.Intensity = 0.075
    SunRays.Spread = 0.727

    Sky.SkyboxBk = "http://www.roblox.com/asset/?id=151165214"
    Sky.SkyboxDn = "http://www.roblox.com/asset/?id=151165197"
    Sky.SkyboxFt = "http://www.roblox.com/asset/?id=151165224"
    Sky.SkyboxLf = "http://www.roblox.com/asset/?id=151165191"
    Sky.SkyboxRt = "http://www.roblox.com/asset/?id=151165206"
    Sky.SkyboxUp = "http://www.roblox.com/asset/?id=151165227"

    Lighting.Ambient = Color3.fromRGB(2, 2, 2)
    Lighting.Brightness = 2.25
    Lighting.ClockTime = 17
    Lighting.ExposureCompensation = 0.5

    Atm.Density = 0.364
    Atm.Offset = 0.556
    Atm.Color = Color3.fromRGB(199, 175, 166)
    Atm.Decay = Color3.fromRGB(44, 39, 33)
    Atm.Glare = 0.36
    Atm.Haze = 1.72
    
    return {Bloom = Bloom, Blur = Blur, ColorCor = ColorCor, SunRays = SunRays, Sky = Sky, Atm = Atm}
end

local function CreateOverlay(texture, transparency, size, position)
    if not overlayGui then
        overlayGui = Instance.new("ScreenGui")
        overlayGui.Name = "OverlayEffects"
        overlayGui.Parent = LocalPlayer.PlayerGui
        overlayGui.ResetOnSpawn = false
    end
    local img = Instance.new("ImageLabel")
    img.Size = size or UDim2.new(1, 0, 1, 0)
    img.Position = position or UDim2.new(0, 0, 0, 0)
    img.BackgroundTransparency = 1
    img.Image = texture or "rbxassetid://4576475446"
    img.ImageTransparency = transparency or 0.3
    img.ZIndex = 10
    img.Parent = overlayGui

end

local function RemoveExtraEffect(name)
    if extraEffects[name] then
        if extraEffects[name]:IsA("ImageLabel") or extraEffects[name]:IsA("Frame") then
            extraEffects[name]:Destroy()
        else
            pcall(function() extraEffects[name]:Destroy() end)
        end
        extraEffects[name] = nil
    end
end

-- ============================================
-- ФУНКЦИЯ KORBLOX (скелетные ноги)
-- ============================================
local function ApplyKorblox(char, enable)
    if not char then return end
    local isR15 = char:FindFirstChild("RightUpperLeg") ~= nil
    if not isR15 then
        warn("Korblox работает только для R15")
        return
    end
    
    local parts = {
        RightLowerLeg = {id = "rbxassetid://902942093"},
        RightFoot = {id = "rbxassetid://902942089"},
        RightUpperLeg = {id = "rbxassetid://902942110"}
    }
    
    -- Если включаем
    if enable then
        -- Сохраняем оригинальные MeshId, если ещё не сохранены
        if not extraEffects.KorbloxOriginal then
            extraEffects.KorbloxOriginal = {}
            for partName, data in pairs(parts) do
                local part = char:FindFirstChild(partName)
                if part then
                    extraEffects.KorbloxOriginal[partName] = part.MeshId
                end
            end
        end
        -- Применяем новые MeshId
        for partName, data in pairs(parts) do
            local part = char:FindFirstChild(partName)
            if part then
                part.MeshId = data.id
            end
        end
    else
        -- Восстанавливаем оригинальные MeshId
        if extraEffects.KorbloxOriginal then
            for partName, originalId in pairs(extraEffects.KorbloxOriginal) do
                local part = char:FindFirstChild(partName)
                if part then
                    part.MeshId = originalId
                end
            end
            extraEffects.KorbloxOriginal = nil
        else
            -- Если не сохраняли, сбрасываем на стандартные (удаляем MeshId)
            for partName, data in pairs(parts) do
                local part = char:FindFirstChild(partName)
                if part then
                    part.MeshId = ""
                end
            end
        end
    end
end

-- Следим за перерождением персонажа, чтобы автоматически применять Korblox, если включен
local function OnCharacterAdded(char)

        -- Для простоты мы добавим обработку в самой функции головы через CharacterAdded
    end
end

    B.MouseButton1Click:Connect(function()
        active = not active
        if active then
            B.Text = "  " .. text .. " [ВКЛ]"
            B.BackgroundColor3 = Color3.fromRGB(0, 60, 50)

            B.TextColor3 = Color3.fromRGB(180, 180, 180)
        end
        pcall(function() callback(active) end)
    end)
    return B
end

local function CreateSlider(text, min, max, default, callback)
    local Container = Instance.new("Frame")
    Container.Size = UDim2.new(1, 0, 0, 50)
    Container.BackgroundTransparency = 1
    Container.Parent = Scroll
    
    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.new(0.5, 0, 0, 20)
    Label.BackgroundTransparency = 1
    Label.Text = text .. ": " .. default
    Label.TextColor3 = Color3.fromRGB(200, 200, 200)
    Label.Font = Enum.Font.Code
    Label.TextSize = 12
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.Parent = Container
    
    local Slider = Instance.new("Frame")
    Slider.Size = UDim2.new(0.8, 0, 0, 6)
    Slider.Position = UDim2.new(0, 0, 0, 25)
    Slider.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
    Slider.BorderSizePixel = 0
    Slider.Parent = Container
    
    local SliderCorner = Instance.new("UICorner")
    SliderCorner.CornerRadius = UDim.new(0, 3)
    SliderCorner.Parent = Slider
    
    local Fill = Instance.new("Frame")
    Fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    Fill.BackgroundColor3 = Color3.fromRGB(0, 255, 200)
    Fill.BorderSizePixel = 0
    Fill.Parent = Slider
    
    local FillCorner = Instance.new("UICorner")
    FillCorner.CornerRadius = UDim.new(0, 3)
    FillCorner.Parent = Fill
    
    local Value = default
    local dragging = false
    
    local function UpdateSlider(mouseX)
        local relX = math.clamp((mouseX - Slider.AbsolutePosition.X) / Slider.AbsoluteSize.X, 0, 1)
        Fill.Size = UDim2.new(relX, 0, 1, 0)
        Value = min + (max - min) * relX
        Value = math.round(Value * 100) / 100
        Label.Text = text .. ": " .. Value
        callback(Value)
    end
    
    Slider.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            UpdateSlider(input.Position.X)
        end
    end)
    
    Slider.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            UpdateSlider(input.Position.X)
        end
    end)
end

-- 16. 👤 БЕЗ ГОЛОВЫ (HEADLESS)
CreateToggle("👤 Без головы", function(on)
    headlessActive = on
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    if not char then return end
    local head = char:FindFirstChild("Head")
    if not head then return end
    
    if on then
        -- Сохраняем исходное состояние (если ещё не сохранили)
        if not extraEffects.HeadlessOriginal then
            extraEffects.HeadlessOriginal = {
                HeadTransparency = head.Transparency,
                Decals = {}
            }
            for _, child in pairs(head:GetChildren()) do
                if child:IsA("Decal") then
                    table.insert(extraEffects.HeadlessOriginal.Decals, {
                        object = child,
                        transparency = child.Transparency
                    })
                end
            end
        end
        -- Делаем голову прозрачной
        head.Transparency = 1
        for _, child in pairs(head:GetChildren()) do
            if child:IsA("Decal") then
                child.Transparency = 1
            end
        end
    else
        -- Восстанавливаем
        if extraEffects.HeadlessOriginal then
            head.Transparency = extraEffects.HeadlessOriginal.HeadTransparency or 0
            for _, decalInfo in pairs(extraEffects.HeadlessOriginal.Decals or {}) do
                if decalInfo.object then
                    decalInfo.object.Transparency = decalInfo.transparency
                end
            end
            extraEffects.HeadlessOriginal = nil
        else
            head.Transparency = 0
            for _, child in pairs(head:GetChildren()) do
                if child:IsA("Decal") then
                    child.Transparency = 0
                end
            end
        end
    end
end)

CreateToggle("🔄 СБРОС ВСЕХ НАСТРОЕК", function(on)
    if on then
        -- Удаляем все эффекты из Lighting
        for i, v in pairs(Lighting:GetChildren()) do
            if v:IsA("BloomEffect") or v:IsA("BlurEffect") or v:IsA("ColorCorrectionEffect") or 
               v:IsA("SunRaysEffect") or v:IsA("Sky") or v:IsA("Atmosphere") then
                v:Destroy()
            end
        end
        -- Сбрасываем настройки
        Lighting.Ambient = Color3.fromRGB(127, 127, 127)
        Lighting.Brightness = 2
        Lighting.ClockTime = 12
        Lighting.ExposureCompensation = 0
        -- Удаляем все дополнительные эффекты (кроме HeadlessOriginal и KorbloxOriginal, их обрабатываем отдельно)
        for name, obj in pairs(extraEffects) do
            if name == "HeadlessOriginal" then
                -- Восстанавливаем голову
                local char = LocalPlayer.Character
                if char then
                    local head = char:FindFirstChild("Head")
                    if head then
                        head.Transparency = obj.HeadTransparency or 0
                        for _, decalInfo in pairs(obj.Decals or {}) do
                            if decalInfo.object then
                                decalInfo.object.Transparency = decalInfo.transparency
                            end
                        end
                    end
                end
            elseif name == "KorbloxOriginal" then
                -- Восстанавливаем ноги Korblox
                local char = LocalPlayer.Character
                if char then
                    for partName, originalId in pairs(obj) do
                        local part = char:FindFirstChild(partName)
                        if part then
                            part.MeshId = originalId
                        end
                    end
                end
            else
                pcall(function() obj:Destroy() end)
            end
        end
        extraEffects = {}
        korbloxActive = false
        headlessActive = false
        if overlayGui then
            overlayGui:Destroy()
            overlayGui = nil
        end
        effects = {}
        -- Обновляем кнопки в GUI
        for _, btn in pairs(Scroll:GetChildren()) do
            if btn:IsA("TextButton") then
                if btn.Text:find("[ВКЛ]") then
                    btn.Text = btn.Text:gsub("%[ВКЛ%]", "[ВЫКЛ]")
                    btn.BackgroundColor3 = Color3.fromRGB(25, 25, 35)
                    btn.TextColor3 = Color3.fromRGB(180, 180, 180)
                end
            end
        end
    end
end)
