 local function upd()
  for _,c in ipairs(l:GetChildren()) do if c:IsA("TextButton") then c:Destroy() end end
  local y=0
  for _,pl in ipairs(game.Players:GetPlayers()) do
   if pl~=p then
    local b=Instance.new("TextButton")
    b.Size=UDim2.new(1,0,0,30)
    b.Position=UDim2.new(0,0,0,y)
    b.BackgroundColor3=Color3.fromRGB(55,55,60)
    b.BorderSizePixel=0
    b.Text=pl.Name
    b.TextColor3=Color3.fromRGB(255,255,255)
    b.TextScaled=true
    b.Font=Enum.Font.Gotham
    b.Parent=l
    b.MouseButton1Click:Connect(function()
     if pl.Character and pl.Character:FindFirstChild("HumanoidRootPart") then
      pl.Character.HumanoidRootPart.CFrame=CFrame.new(0,-500,0)
      print("🚀 "..pl.Name.." выкинут!")
     end
    end)
    y=y+30
   end
  end
 end
 upd()
 game.Players.PlayerAdded:Connect(upd)
 game.Players.PlayerRemoving:Connect(upd)
end
