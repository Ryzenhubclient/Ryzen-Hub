-- Загружаем оригинальный скрипт KitagawaHub
local kitagawaScript = loadstring(game:HttpGet("https://raw.githubusercontent.com/Terfiscript1/KitagawaHub/refs/heads/main/kitagawahub"))

if type(kitagawaScript) == "function" then
    -- Запускаем оригинальный скрипт
    local success, err = pcall(kitagawaScript)
    if not success then
        warn("Ошибка при загрузке KitagawaHub: ", err)
        return
    end

    -- Даём время на создание GUI
    task.wait(1)

    -- Функция для замены названия и логотипа
    local function replaceKitagawaWithRyzen(obj)
        -- Меняем название объекта
        if obj.Name and obj.Name:lower():find("kitagawa") then
            obj.Name = string.gsub(obj.Name, "[Kk][Ii][Tt][Aa][Gg][Aa][Ww][Aa]", "RyzenHub")
        end

        -- Меняем текст на всех текстовых элементах
        if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
            if obj.Text and obj.Text:lower():find("kitagawa") then
                obj.Text = string.gsub(obj.Text, "[Kk][Ii][Tt][Aa][Gg][Aa][Ww][Aa]", "Ryzen Hub")
            end
        end

        -- Меняем изображение (если это ImageLabel или ImageButton)
        if obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
            -- Проверяем, похоже ли изображение на логотип (капучино, асасин и т.д.)
            if obj.Image and (obj.Image:lower():find("cap") or obj.Image:lower():find("capp") or obj.Image:lower():find("assassin") or obj.Image:lower():find("logo")) then
                -- Заменяем на логотип Ryzen (ссылка на изображение с википедии)
                obj.Image = "https://upload.wikimedia.org/wikipedia/commons/thumb/7/7f/AMD_Ryzen_logo.svg/120px-AMD_Ryzen_logo.svg.png"
            end
        end

        -- Рекурсивно проходим по всем дочерним элементам
        for _, child in ipairs(obj:GetChildren()) do
            replaceKitagawaWithRyzen(child)
        end
    end

    -- Ищем все GUI и применяем замену
    local replaced = false
    for _, gui in ipairs(game.CoreGui:GetChildren()) do
        if gui:IsA("ScreenGui") then
            replaceKitagawaWithRyzen(gui)
            replaced = true
        end
    end

    -- Если не нашли в CoreGui, пробуем в PlayerGui
    if not replaced then
        for _, gui in ipairs(game.Players.LocalPlayer.PlayerGui:GetChildren()) do
            if gui:IsA("ScreenGui") then
                replaceKitagawaWithRyzen(gui)
                replaced = true
            end
        end
    end

    if replaced then
        print("✅ Замена выполнена: 'Kitagawa' → 'Ryzen Hub', логотип заменён на Ryzen")
    else
        print("⚠️ GUI не найден. Возможно, он создаётся с задержкой.")
    end
else
    warn("Не удалось загрузить KitagawaHub.")
end
