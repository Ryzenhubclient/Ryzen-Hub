-- ===== ОПРЕДЕЛЕНИЕ РОЛИ =====
local function getRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    for _, tool in ipairs(plr.Character:GetChildren()) do
        if tool:IsA("Tool") then
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    if plr.Backpack then
        for _, tool in ipairs(plr.Backpack:GetChildren()) do
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    return "Innocent"
end

-- ===== AUTO LOOTGUN ДЛЯ MM2 =====
local function startLootgun()
    if lootgunConnection then
        lootgunConnection:Disconnect()
        lootgunConnection = nil
    end
    
    if not toggles.lootgun then return end
    
    lootgunConnection = rs.Heartbeat:Connect(function()
        if not player.Character then return end
        
        local character = player.Character
        local root = character:FindFirstChild("HumanoidRootPart")
        if not root then return end
        
        local rootPos = root.Position
        
        -- Ищем все объекты на карте
        for _, v in ipairs(workspace:GetDescendants()) do
            -- Проверяем, является ли объект пистолетом (находим по названию)
            if v:IsA("Tool") then
                local name = v.Name:lower()
                -- Проверяем названия пистолетов в MM2
                if name:find("gun") or name:find("pistol") or name:find("revolver") or name:find("sheriff") then
                    -- Проверяем, что пистолет не у игрока и не в инвентаре
                    if v.Parent ~= character and v.Parent ~= player.Backpack then
                        -- Проверяем расстояние
                        local distance = (v.Parent and v.Parent:IsA("BasePart") and (v.Parent.Position - rootPos).Magnitude) or 999
                        
                        -- Если есть Handle или прикреплен к части
                        local handle = v:FindFirstChild("Handle")
                        if handle then
                            distance = (handle.Position - rootPos).Magnitude
                        end
                        
                        -- Если пистолет в радиусе - забираем
                        if distance <= settings.lootgunRange then
                            pcall(function()
                                -- Проверяем, не держит ли уже игрок оружие
                                local hasGun = false
                                for _, tool in ipairs(character:GetChildren()) do
                                    if tool:IsA("Tool") then
                                        local tName = tool.Name:lower()
                                        if tName:find("gun") or tName:find("pistol") or tName:find("revolver") then
                                            hasGun = true
                                            break
                                        end
                                    end
                                end
                                
                                -- Если у игрока нет пистолета - забираем
                                if not hasGun then
                                    -- Проверяем, что пистолет существует и не у другого игрока
                                    if v.Parent and v.Parent:IsA("Model") and v.Parent:FindFirstChild("Humanoid") then
                                        -- Если пистолет у другого игрока - пропускаем
                                        return
                                    end
                                    
                                    -- Забираем пистолет
                                    v.Parent = character
                                    print("🔫 Пистолет подобран!")
                                    
                                    -- Если есть анимация подбора
                                    local humanoid = character:FindFirstChild("Humanoid")
                                    if humanoid then
                                        humanoid:LoadAnimation(v:FindFirstChild("Animation"))
                                    end
                                end
                            end)
                        end
                    end
                end
            end
        end
    end)
end

-- ===== ОСТАЛЬНЫЕ ФУНКЦИИ (заглушки) =====
function updateESP()
    for _, v in pairs(espObjects) do v:Destroy() end
    espObjects = {}
    if not toggles.esp then return end
    
    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local esp = Instance.new("BillboardGui")
            esp.Size = UDim2.new(0, 200, 0, 50)
            esp.AlwaysOnTop = true
            esp.Parent = plr.Character.HumanoidRootPart
            
            local label = Instance.new("TextLabel")
            label.Size = UDim2.new(1, 0, 1, 0)
            label.BackgroundTransparency = 1
            label.TextColor3 = Color3.fromRGB(255, 0, 0)
            label.TextScaled = true
            label.Font = Enum.Font.GothamBold
            label.Text = plr.Name .. " (" .. getRole(plr) .. ")"
            label.Parent = esp
            
            table.insert(espObjects, esp)
        end
    end
end

-- ===== ФУНКЦИЯ ДЛЯ LOOTGUN =====
function toggleLootgun(s)
    toggles.lootgun = s
    if s then
        startLootgun()
    else
        if lootgunConnection then
            lootgunConnection:Disconnect()
            lootgunConnection = nil
        end
    end
end
