-- ===== НАСТРОЙКИ =====
local settings = {
    speed = 100,
    flyspeed = 50,
    jumppower = 120,
    hitboxSize = 10,
    antiAFK = false,
}

local toggles = {
    esp = false,

local connections = {}

-- ===== ОПРЕДЕЛЕНИЕ РОЛИ =====
local function getRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    for _, tool in ipairs(plr.Character:GetChildren()) do
        if tool:IsA("Tool") then
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    if plr.Backpack then
        for _, tool in ipairs(plr.Backpack:GetChildren()) do
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    return "Innocent"
end

-- ===== ESP =====
local function updateESP()
    for _, obj in ipairs(espObjects) do
        pcall(function() obj:Destroy() end)
    end
    espObjects = {}
    if not toggles.esp then return end

    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr == player then continue end
        if not plr.Character or not plr.Character:FindFirstChild("Head") then continue end
        local role = getRole(plr)
        local color = role == "Murderer" and Color3.fromRGB(255,0,0) or role == "Sheriff" and Color3.fromRGB(0,100,255) or Color3.fromRGB(0,255,0)
        local esp = Instance.new("BillboardGui")
        esp.Size = UDim2.new(0, 150, 0, 40)
        esp.AlwaysOnTop = true
        esp.Parent = plr.Character.Head
        table.insert(espObjects, esp)
        local text = Instance.new("TextLabel")
        text.Size = UDim2.new(1,0,1,0)
        text.BackgroundTransparency = 1
        text.Text = plr.Name .. " [" .. role .. "]"
        text.TextColor3 = color
        text.TextScaled = true
        text.Font = Enum.Font.GothamBold
        text.Parent = esp
        table.insert(espObjects, text)
        local hl = Instance.new("Highlight")
        hl.Adornee = plr.Character
        hl.FillColor = color
        hl.FillTransparency = 0.5
        hl.Parent = plr.Character
        table.insert(espObjects, hl)
    end
end

-- ===== SILENT AIM =====
local function toggleSilentAim(state)
    toggles.silentAim = state
    if state then
        if connections.silentAim then connections.silentAim:Disconnect() end
        connections.silentAim = rs.Heartbeat:Connect(function()
            if not toggles.silentAim then return end
            if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
            local target, targetDist = nil, math.huge
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character and plr.Character:FindFirstChild("Head") then
                    local dist = (plr.Character.Head.Position - player.Character.HumanoidRootPart.Position).Magnitude
                    if dist < targetDist and dist < 150 then
                        target = plr
                        targetDist = dist
                    end
                end
            end
            if target then
                player.Character.HumanoidRootPart.CFrame = CFrame.new(player.Character.HumanoidRootPart.Position, target.Character.Head.Position)
            end
        end)
    else
        if connections.silentAim then connections.silentAim:Disconnect() end
    end
end

-- ===== HITBOX EXTENDER =====
local function toggleHitbox(state)
    toggles.hitbox = state
    if state then
        if connections.hitbox then connections.hitbox:Disconnect() end
        connections.hitbox = rs.Heartbeat:Connect(function()
            if not toggles.hitbox then return end
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character then
                    for _, part in ipairs(plr.Character:GetChildren()) do
                        if part:IsA("BasePart") then
                            part.Size = part.Size + Vector3.new(settings.hitboxSize, settings.hitboxSize, settings.hitboxSize)
                        end
                    end
                end
            end
        end)
    else
        if connections.hitbox then connections.hitbox:Disconnect() end
        for _, plr in ipairs(game.Players:GetPlayers()) do
            if plr ~= player and plr.Character then
                for _, part in ipairs(plr.Character:GetChildren()) do
                    if part:IsA("BasePart") then
                        part.Size = Vector3.new(2, 2, 1)
                    end
                end
            end
        end
    end
end

-- ===== ANTI-AFK =====

    toggles.antiAFK = state
    if state then

            if not toggles.antiAFK then return end

end

-- ===== AUTO COLLECT (МОНЕТЫ) =====
local function toggleAutoCollect(state)
    toggles.autoCollect = state
    if state then
        if connections.autoCollect then connections.autoCollect:Disconnect() end
        connections.autoCollect = rs.Heartbeat:Connect(function()
            if not toggles.autoCollect then return end
            if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
            local root = player.Character.HumanoidRootPart
            for _, item in ipairs(workspace:GetDescendants()) do
                if item:IsA("Part") and item.Name:lower():find("coin") then
                    local dist = (item.Position - root.Position).Magnitude
                    if dist < 30 then
                        root.CFrame = CFrame.new(item.Position + Vector3.new(0, 2, 0))
                        task.wait(0.1)
                    end
                end
            end
        end)
    else
        if connections.autoCollect then connections.autoCollect:Disconnect() end
    end
end

-- ===== WALL HOP =====
local function toggleWallHop(state)
    toggles.wallHop = state
    if state then
        if connections.wallHop then connections.wallHop:Disconnect() end
        connections.wallHop = rs.Stepped:Connect(function()
            if not toggles.wallHop then return end
            if player.Character and player.Character:FindFirstChild("Humanoid") then
                local h = player.Character.Humanoid
                if h:GetState() == Enum.HumanoidStateType.Jumping or h:GetState() == Enum.HumanoidStateType.Falling then
                    h.WalkSpeed = 200
                else
                    h.WalkSpeed = 16
                end
            end
        end)
    else
        if connections.wallHop then connections.wallHop:Disconnect() end
        if player.Character and player.Character:FindFirstChild("Humanoid") then
            player.Character.Humanoid.WalkSpeed = 16
        end
    end
end

-- ===== TELEPORT TO PLAYER =====
local function toggleTeleportToPlayer(state)
    toggles.teleportToPlayer = state
end

local function teleportToPlayer()
    if not toggles.teleportToPlayer then return end
    local target = nil
    local targetDist = math.huge
    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local dist = (plr.Character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
            if dist < targetDist then
                target = plr
                targetDist = dist
            end
        end
    end
    if target and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
        player.Character.HumanoidRootPart.CFrame = target.Character.HumanoidRootPart.CFrame * CFrame.new(0, 2, 0)
    end
end

-- Кнопка T для телепорта к игроку
uis.InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.T and toggles.teleportToPlayer then
        teleportToPlayer()
    end
end)

-- ===== FLING PLAYER =====
local function toggleFlingPlayer(state)
    toggles.flingPlayer = state
    if state then
        if connections.flingPlayer then connections.flingPlayer:Disconnect() end
        connections.flingPlayer = rs.Stepped:Connect(function()
            if not toggles.flingPlayer then return end
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                for _, plr in ipairs(game.Players:GetPlayers()) do
                    if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                        local dist = (plr.Character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
                        if dist < 10 then
                            pcall(function()
                                plr.Character.HumanoidRootPart.Velocity = Vector3.new(0, 1000, 0)
                            end)
                        end
                    end
                end
            end
        end)
    else
        if connections.flingPlayer then connections.flingPlayer:Disconnect() end
    end
end
