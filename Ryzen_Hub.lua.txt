-- RYZEN HUB - МИНИМАЛЬНАЯ ВЕРСИЯ ДЛЯ XENO
local plr = game.Players.LocalPlayer
local mouse = plr:GetMouse()

-- Настройки
local speed = 100
local flySpeed = 50
local jumpPower = 120

-- Состояния
local espOn = false
local triggerOn = false
local speedOn = false
local jumpOn = false
local noclipOn = false
local flyOn = false
local teleOn = false
local shotOn = false

-- Список для ESP
local espObjs = {}

-- Функция роли
local function getRole(p)
    if not p or not p.Character then return "Innocent" end
    for _, t in ipairs(p.Character:GetChildren()) do
        if t:IsA("Tool") then
            local n = t.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    if p.Backpack then
        for _, t in ipairs(p.Backpack:GetChildren()) do
            if t:IsA("Tool") then
                local n = t.Name:lower()
                if n:find("knife") or n:find("dagger") then return "Murderer" end
                if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
            end
        end
    end
    return "Innocent"
end

-- ESP
local function updateESP()
    for _, o in ipairs(espObjs) do
        pcall(function() o:Destroy() end)
    end
    espObjs = {}
    if not espOn then return end
    for _, p in ipairs(game.Players:GetPlayers()) do
        if p == plr then continue end
        if not p.Character or not p.Character:FindFirstChild("Head") then continue end
        local role = getRole(p)
        local col = role == "Murderer" and Color3.fromRGB(255,0,0) or role == "Sheriff" and Color3.fromRGB(0,100,255) or Color3.fromRGB(0,255,0)
        local bill = Instance.new("BillboardGui")
        bill.Size = UDim2.new(0,150,0,40)
        bill.AlwaysOnTop = true
        bill.Parent = p.Character.Head
        local txt = Instance.new("TextLabel")
        txt.Size = UDim2.new(1,0,1,0)
        txt.BackgroundTransparency = 1
        txt.Text = p.Name .. " [" .. role .. "]"
        txt.TextColor3 = col
        txt.TextScaled = true
        txt.Font = Enum.Font.GothamBold
        txt.Parent = bill
        table.insert(espObjs, bill)
        table.insert(espObjs, txt)
        local hl = Instance.new("Highlight")
        hl.Adornee = p.Character
        hl.FillColor = col
        hl.FillTransparency = 0.5
        hl.Parent = p.Character
        table.insert(espObjs, hl)
    end
end

-- Trigger Bot
local trigConn = nil
local function toggleTrigger(state)
    triggerOn = state
    if state then
        if trigConn then trigConn:Disconnect() end
        trigConn = game:GetService("RunService").Heartbeat:Connect(function()
            if not triggerOn then return end
            for _, p in ipairs(game.Players:GetPlayers()) do
                if p ~= plr and p.Character and p.Character:FindFirstChild("Humanoid") then
                    if getRole(p) == "Murderer" then
                        pcall(function() p.Character.Humanoid.Health = 0 end)
                    end
                end
            end
        end)
    else
        if trigConn then trigConn:Disconnect() end
    end
end

-- Speed Glitch
local speedConn = nil
local function toggleSpeed(state)
    speedOn = state
    if state then
        if speedConn then speedConn:Disconnect() end
        speedConn = game:GetService("RunService").Stepped:Connect(function()
            if not speedOn then return end
            if plr.Character and plr.Character:FindFirstChild("Humanoid") then
                local h = plr.Character.Humanoid
                if h:GetState() == Enum.HumanoidStateType.Jumping or h:GetState() == Enum.HumanoidStateType.Falling then
                    h.WalkSpeed = speed
                else
                    h.WalkSpeed = 16
                end
            end
        end)
    else
        if speedConn then speedConn:Disconnect() end
        if plr.Character and plr.Character:FindFirstChild("Humanoid") then
            plr.Character.Humanoid.WalkSpeed = 16
        end
    end
end

-- Super Jump
local function toggleJump(state)
    jumpOn = state
    if plr.Character and plr.Character:FindFirstChild("Humanoid") then
        plr.Character.Humanoid.JumpPower = state and jumpPower or 50
    end
end

-- Noclip
local noclipConn = nil
local function toggleNoclip(state)
    noclipOn = state
    if state then
        if noclipConn then noclipConn:Disconnect() end
        noclipConn = game:GetService("RunService").Stepped:Connect(function()
            if not noclipOn then return end
            if plr.Character then
                for _, part in ipairs(plr.Character:GetChildren()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end)
    else
        if noclipConn then noclipConn:Disconnect() end
        if plr.Character then
            for _, part in ipairs(plr.Character:GetChildren()) do
                if part:IsA("BasePart") then part.CanCollide = true end
            end
        end
    end
end

-- Fly
local flyConn = nil
local function toggleFly(state)
    flyOn = state
    if state then
        if flyConn then flyConn:Disconnect() end
        flyConn = game:GetService("RunService").Stepped:Connect(function()
            if not flyOn then return end
            if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                local root = plr.Character.HumanoidRootPart
                local dir = Vector3.new(0,0,0)
                local uis = game:GetService("UserInputService")
                if uis:IsKeyDown(Enum.KeyCode.W) then dir = dir + root.CFrame.LookVector end
                if uis:IsKeyDown(Enum.KeyCode.S) then dir = dir - root.CFrame.LookVector end
                if uis:IsKeyDown(Enum.KeyCode.A) then dir = dir - root.CFrame.RightVector end
                if uis:IsKeyDown(Enum.KeyCode.D) then dir = dir + root.CFrame.RightVector end
                if uis:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
                if uis:IsKeyDown(Enum.KeyCode.LeftShift) then dir = dir - Vector3.new(0,1,0) end
                root.Velocity = dir * flySpeed
            end
        end)
    else
        if flyConn then flyConn:Disconnect() end
        if plr.Character and plr.Character:FindFirstChild("Humanoid") then
            plr.Character.Humanoid.PlatformStand = false
        end
    end
end

-- Teleport
local function toggleTeleport(state)
    teleOn = state
end
mouse.Button1Down:Connect(function()
    if teleOn and mouse.Target then
        local pos = mouse.Target.Position
        if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            plr.Character.HumanoidRootPart.CFrame = CFrame.new(pos)
        end
    end
end)

-- Shot Murderer
local function toggleShot(state)
    shotOn = state
end
local function killMurderer()
    if not shotOn then return end
    for _, p in ipairs(game.Players:GetPlayers()) do
        if p ~= plr and p.Character and p.Character:FindFirstChild("Humanoid") then
            if getRole(p) == "Murderer" then
                pcall(function() p.Character.Humanoid.Health = 0 end)
                return
            end
        end
    end
end

-- Забрать пистолет
local function pickupGun()
    if not plr.Character or not plr.Character:FindFirstChild("HumanoidRootPart") then return end
    local root = plr.Character.HumanoidRootPart
    for _, item in ipairs(workspace:GetDescendants()) do
        if item:IsA("Tool") then
            local n = item.Name:lower()
            if n:find("gun") or n:find("pistol") or n:find("revolver") then
                if item.Parent ~= plr.Character and item.Parent ~= plr.Backpack then
                    local dist = (item.Position - root.Position).Magnitude
                    if dist < 100 then
                        pcall(function()
                            root.CFrame = CFrame.new(item.Position + Vector3.new(0,2,0))
                            task.wait(0.2)
                            item.Parent = plr.Character
                        end)
                        return
                    end
                end
            end
        end
    end
end

-- GUI
local gui = nil
local function createGUI()
    if gui then gui:Destroy() end
    gui = Instance.new("ScreenGui")
    gui.Name = "RyzenHub"
    gui.Parent = game.CoreGui
    gui.ResetOnSpawn = false

    local main = Instance.new("Frame")
    main.Size = UDim2.new(0, 350, 0, 460)
    main.Position = UDim2.new(0.5, -175, 0.5, -230)
    main.BackgroundColor3 = Color3.fromRGB(15,0,15)
    main.BackgroundTransparency = 0.2
    main.BorderSizePixel = 2
    main.BorderColor3 = Color3.fromRGB(200,0,255)
    main.Active = true
    main.Draggable = true
    main.Parent = gui

    local title = Instance.new("Frame")
    title.Size = UDim2.new(1,0,0,35)
    title.BackgroundColor3 = Color3.fromRGB(30,0,30)
    title.BorderSizePixel = 0
    title.Parent = main

    local logo = Instance.new("ImageLabel")
    logo.Size = UDim2.new(0,25,0,25)
    logo.Position = UDim2.new(0.02,0,0.1,0)
    logo.BackgroundTransparency = 1
    logo.Image = "https://upload.wikimedia.org/wikipedia/commons/thumb/7/7f/AMD_Ryzen_logo.svg/120px-AMD_Ryzen_logo.svg.png"
    logo.Parent = title

    local txt = Instance.new("TextLabel")
    txt.Size = UDim2.new(1,-45,1,0)
    txt.Position = UDim2.new(0,45,0,0)
    txt.BackgroundTransparency = 1
    txt.Text = "RYZEN HUB"
    txt.TextColor3 = Color3.fromRGB(255,0,200)
    txt.TextScaled = true
    txt.Font = Enum.Font.GothamBold
    txt.Parent = title

    local close = Instance.new("TextButton")
    close.Size = UDim2.new(0,25,0,25)
    close.Position = UDim2.new(1,-30,0,5)
    close.BackgroundColor3 = Color3.fromRGB(80,0,80)
    close.BorderSizePixel = 0
    close.Text = "✕"
    close.TextColor3 = Color3.fromRGB(255,255,255)
    close.TextScaled = true
    close.Font = Enum.Font.GothamBold
    close.Parent = title
    close.MouseButton1Click:Connect(function() gui:Destroy() end)

    local y = 0.08
    local function addBtn(text, cb)
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(0.85,0,0,26)
        btn.Position = UDim2.new(0.075,0,y,0)
        btn.BackgroundColor3 = Color3.fromRGB(30,0,30)
        btn.BorderSizePixel = 1
        btn.BorderColor3 = Color3.fromRGB(200,0,255)
        btn.Text = text
        btn.TextColor3 = Color3.fromRGB(255,255,255)
        btn.TextScaled = true
        btn.Font = Enum.Font.Gotham
        btn.Parent = main
        local active = false
        btn.MouseButton1Click:Connect(function()
            active = not active
            if active then
                btn.BackgroundColor3 = Color3.fromRGB(80,0,80)
                btn.TextColor3 = Color3.fromRGB(255,200,255)
            else
                btn.BackgroundColor3 = Color3.fromRGB(30,0,30)
                btn.TextColor3 = Color3.fromRGB(255,255,255)
            end
            cb(active)
            if text:find("ESP") then updateESP() end
        end)
        y = y + 0.07
        return btn
    end

    local function addSlider(txt, min, max, def, cb)
        local fr = Instance.new("Frame")
        fr.Size = UDim2.new(0.85,0,0,22)
        fr.Position = UDim2.new(0.075,0,y,0)
        fr.BackgroundColor3 = Color3.fromRGB(20,0,20)
        fr.BorderSizePixel = 0
        fr.Parent = main

        local lbl = Instance.new("TextLabel")
        lbl.Size = UDim2.new(0.4,0,1,0)
        lbl.Position = UDim2.new(0.02,0,0,0)
        lbl.BackgroundTransparency = 1
        lbl.Text = txt
        lbl.TextColor3 = Color3.fromRGB(200,150,255)
        lbl.TextScaled = true
        lbl.Font = Enum.Font.Gotham
        lbl.Parent = fr

        local sl = Instance.new("TextButton")
        sl.Size = UDim2.new(0.4,0,0.7,0)
        sl.Position = UDim2.new(0.45,0,0.15,0)
        sl.BackgroundColor3 = Color3.fromRGB(40,0,40)
        sl.BorderSizePixel = 1
        sl.BorderColor3 = Color3.fromRGB(200,0,255)
        sl.Text = tostring(def)
        sl.TextColor3 = Color3.fromRGB(255,255,255)
        sl.TextScaled = true
        sl.Font = Enum.Font.Gotham
        sl.Parent = fr
        local val = def
        sl.MouseButton1Click:Connect(function()
            val = val + 10
            if val > max then val = min end
            sl.Text = tostring(val)
            cb(val)
        end)
        y = y + 0.065
        return sl
    end

    -- Кнопки
    addBtn("👁️ ESP", function(s) espOn = s updateESP() end)
    addBtn("🎯 Trigger Bot", toggleTrigger)
    addBtn("⚡ Speed Glitch", toggleSpeed)
    addBtn("🦘 Super Jump", toggleJump)
    addBtn("🌀 Noclip", toggleNoclip)
    addBtn("📍 Teleport", toggleTeleport)
    addBtn("✈️ Fly", toggleFly)
    addBtn("🔫 Shot Murderer [H]", toggleShot)
    addBtn("🔫 Забрать пистолет [G]", function() pickupGun() end)

    -- Слайдеры
    addSlider("⚡ Speed", 20, 300, speed, function(v) speed = v end)
    addSlider("✈️ Fly Speed", 10, 200, flySpeed, function(v) flySpeed = v end)
    addSlider("🦘 Jump Power", 50, 300, jumpPower, function(v) jumpPower = v end)

    print("✅ Ryzen Hub загружен!")
end

-- Открытие по правому Shift
game:GetService("UserInputService").InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.RightShift then
        if gui and gui.Parent then
            gui:Destroy()
            gui = nil
        else
            createGUI()
        end
    end
end)

-- Клавиши H и G
game:GetService("UserInputService").InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.H then
        killMurderer()
    end
    if input.KeyCode == Enum.KeyCode.G then
        pickupGun()
    end
end)

createGUI()
print("✅ RYZEN HUB готов! Нажми ПРАВЫЙ SHIFT.")
