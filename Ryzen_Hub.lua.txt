    silentAim = false,

    antiAFK = false,
    autoCollect = false,
    killAll = false,
    wallHop = false,
    speedGlitch = false,

local miniMenu = nil

local function createMiniMenu(funcName, x, y)
    if miniMenu then miniMenu:Destroy() end
    miniMenu = Instance.new("ScreenGui")
    miniMenu.Name = "MiniMenu"
    miniMenu.Parent = game.CoreGui

    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(0, 300, 0, 350) -- Увеличенный размер
    frame.Position = UDim2.new(0, x, 0, y)
    frame.BackgroundColor3 = Color3.fromRGB(20, 0, 0)
    frame.BackgroundTransparency = 0.1
    frame.BorderSizePixel = 2
    frame.BorderColor3 = Color3.fromRGB(255, 0, 0)
    frame.Active = true
    frame.Draggable = true
    frame.Parent = miniMenu

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1,0,0,30)
    title.BackgroundColor3 = Color3.fromRGB(40, 0, 0)
    title.BorderSizePixel = 0
    title.Text = "⚙️ " .. funcName
    title.TextColor3 = Color3.fromRGB(255, 50, 50)
    title.TextScaled = true
    title.Font = Enum.Font.GothamBold
    title.Parent = frame

    local close = Instance.new("TextButton")
    close.Size = UDim2.new(0,25,0,25)
    close.Position = UDim2.new(1,-30,0,2)
    close.BackgroundColor3 = Color3.fromRGB(80,0,0)
    close.BorderSizePixel = 0
    close.Text = "✕"
    close.TextColor3 = Color3.fromRGB(255,255,255)
    close.TextScaled = true
    close.Font = Enum.Font.GothamBold
    close.Parent = frame
    close.MouseButton1Click:Connect(function() miniMenu:Destroy() end)

    -- ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ ДЛЯ ЭЛЕМЕНТОВ МЕНЮ
    local function createSlider(txt, y, min, max, def, cb)
        local fr = Instance.new("Frame")
        fr.Size = UDim2.new(0.9,0,0,25)
        fr.Position = UDim2.new(0.05,0,y,0)
        fr.BackgroundColor3 = Color3.fromRGB(30,0,0)
        fr.BorderSizePixel = 0
        fr.Parent = frame

        local lbl = Instance.new("TextLabel")
        lbl.Size = UDim2.new(0.4,0,1,0)
        lbl.Position = UDim2.new(0.02,0,0,0)
        lbl.BackgroundTransparency = 1
        lbl.Text = txt
        lbl.TextColor3 = Color3.fromRGB(255,200,200)
        lbl.TextScaled = true
        lbl.Font = Enum.Font.Gotham
        lbl.Parent = fr

        local sl = Instance.new("TextButton")
        sl.Size = UDim2.new(0.4,0,0.7,0)
        sl.Position = UDim2.new(0.45,0,0.15,0)
        sl.BackgroundColor3 = Color3.fromRGB(50,0,0)
        sl.BorderSizePixel = 1
        sl.BorderColor3 = Color3.fromRGB(255,50,50)
        sl.Text = tostring(def)
        sl.TextColor3 = Color3.fromRGB(255,255,255)
        sl.TextScaled = true
        sl.Font = Enum.Font.Gotham
        sl.Parent = fr
        local val = def
        sl.MouseButton1Click:Connect(function()
            val = val + 10
            if val > max then val = min end
            sl.Text = tostring(val)
            cb(val)
        end)
        return sl
    end

    local function createBtn(text, y, cb)
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(0.8,0,0,25)
        btn.Position = UDim2.new(0.1,0,y,0)
        btn.BackgroundColor3 = Color3.fromRGB(40,0,0)
        btn.BorderSizePixel = 1
        btn.BorderColor3 = Color3.fromRGB(255,50,50)
        btn.Text = text
        btn.TextColor3 = Color3.fromRGB(255,255,255)
        btn.TextScaled = true
        btn.Font = Enum.Font.Gotham
        btn.Parent = frame
        btn.MouseButton1Click:Connect(cb)
        return btn
    end

    local function createToggle(text, y, def, cb)
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(0.8,0,0,25)
        btn.Position = UDim2.new(0.1,0,y,0)
        btn.BackgroundColor3 = Color3.fromRGB(40,0,0)
        btn.BorderSizePixel = 1
        btn.BorderColor3 = Color3.fromRGB(255,50,50)
        btn.Text = text .. ": " .. (def and "ON" or "OFF")
        btn.TextColor3 = Color3.fromRGB(255,255,255)
        btn.TextScaled = true
        btn.Font = Enum.Font.Gotham
        btn.Parent = frame
        local state = def
        btn.MouseButton1Click:Connect(function()
            state = not state
            btn.Text = text .. ": " .. (state and "ON" or "OFF")
            cb(state)
        end)
        return btn
    end

    -- ===== НАСТРОЙКИ ДЛЯ КАЖДОЙ ФУНКЦИИ =====
    local yPos = 0.12

    if funcName == "ESP" then
        createBtn("Режим: " .. settings.espMode, yPos, function()
            local modes = {"Имена", "Роли", "Расстояние"}
            local idx = table.find(modes, settings.espMode) or 1
            settings.espMode = modes[idx % #modes + 1]
            btn.Text = "Режим: " .. settings.espMode
            updateESP()
        end)
        yPos = yPos + 0.1
        createSlider("Прозрачность", yPos, 1, 10, settings.espTransparency * 10, function(v) 
            settings.espTransparency = v / 10 
            updateESP()
        end)
        yPos = yPos + 0.1
        createToggle("Показывать дистанцию", yPos, true, function(s) settings.espShowDistance = s updateESP() end)
        yPos = yPos + 0.1
        createToggle("Показывать здоровье", yPos, false, function(s) settings.espShowHealth = s updateESP() end)
        yPos = yPos + 0.1
        createBtn("Сбросить ESP", yPos, function() 
            for _, obj in pairs(espObjects) do obj:Destroy() end
            espObjects = {}
        end)

    elseif funcName == "Silent Aim" then
        createBtn("Цель: " .. settings.silentAimTarget, yPos, function()
            local targets = {"All", "Murderer", "Sheriff", "Innocent"}
            local idx = table.find(targets, settings.silentAimTarget) or 1
            settings.silentAimTarget = targets[idx % #targets + 1]
            btn.Text = "Цель: " .. settings.silentAimTarget
        end)
        yPos = yPos + 0.1
        createSlider("Радиус", yPos, 50, 500, settings.silentAimRange, function(v) settings.silentAimRange = v end)
        yPos = yPos + 0.1
        createToggle("Только голова", yPos, false, function(s) settings.silentAimHeadOnly = s end)
        yPos = yPos + 0.1
        createSlider("Скорость прицела", yPos, 1, 100, 50, function(v) settings.silentAimSpeed = v / 10 end)

    elseif funcName == "Hitbox" then
        createSlider("Размер", yPos, 1, 50, settings.hitboxSize, function(v) 
            settings.hitboxSize = v 
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character then
                    for _, part in ipairs(plr.Character:GetChildren()) do
                        if part:IsA("BasePart") then
                            part.Size = Vector3.new(v, v, v)
                        end
                    end
                end
            end
        end)
        yPos = yPos + 0.1
        createBtn("Сбросить размер", yPos, function()
            settings.hitboxSize = 2
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character then
                    for _, part in ipairs(plr.Character:GetChildren()) do
                        if part:IsA("BasePart") then
                            part.Size = Vector3.new(2, 2, 2)
                        end
                    end
                end
            end
        end)
        yPos = yPos + 0.1
        createToggle("Только голова", yPos, false, function(s) settings.hitboxHeadOnly = s end)

    elseif funcName == "Wall Hop" then
        createSlider("Скорость", yPos, 50, 1000, settings.wallHopSpeed, function(v) settings.wallHopSpeed = v end)
        yPos = yPos + 0.1
        createToggle("Авто-прыжок", yPos, true, function(s) settings.wallHopAutoJump = s end)
        yPos = yPos + 0.1
        createSlider("Высота прыжка", yPos, 1, 20, 10, function(v) settings.wallHopJumpHeight = v end)

    elseif funcName == "Speed Glitch" then
        createSlider("Скорость", yPos, 20, 500, settings.speedGlitch, function(v) settings.speedGlitch = v end)
        yPos = yPos + 0.1
        createToggle("Режим рывка", yPos, false, function(s) settings.speedGlitchDashMode = s end)
        yPos = yPos + 0.1
        createSlider("Длительность", yPos, 1, 30, 10, function(v) settings.speedGlitchDuration = v end)

    elseif funcName == "Auto Collect" then
        createSlider("Радиус", yPos, 10, 200, settings.autoCollectRange, function(v) settings.autoCollectRange = v end)
        yPos = yPos + 0.1
        createSlider("Задержка", yPos, 0.1, 5, settings.autoCollectDelay, function(v) settings.autoCollectDelay = v end)
        yPos = yPos + 0.1
        createToggle("Собирать всё", yPos, true, function(s) settings.autoCollectAll = s end)
        yPos = yPos + 0.1
        createToggle("Игнорировать своих", yPos, false, function(s) settings.autoCollectIgnoreTeam = s end)

    elseif funcName == "Teleport" then
        createBtn("К ближайшему", yPos, function()
            local target = nil
            local dist = math.huge
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                    local d = (plr.Character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
                    if d < dist then
                        target = plr
                        dist = d
                    end
                end
            end
            if target and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                player.Character.HumanoidRootPart.CFrame = target.Character.HumanoidRootPart.CFrame * CFrame.new(0, 2, 0)
            end
        end)
        yPos = yPos + 0.1
        createBtn("К случайному", yPos, function()
            local players = {}
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                    table.insert(players, plr)
                end
            end
            if #players > 0 then
                local target = players[math.random(1, #players)]
                if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    player.Character.HumanoidRootPart.CFrame = target.Character.HumanoidRootPart.CFrame * CFrame.new(0, 2, 0)
                end
            end
        end)
        yPos = yPos + 0.1
        createSlider("Дистанция", yPos, 10, 200, settings.teleportDistance, function(v) settings.teleportDistance = v end)
        yPos = yPos + 0.1
        createToggle("Только к убийце", yPos, false, function(s) settings.teleportToMurderer = s end)

    elseif funcName == "Fling" then
        createBtn("Цель: " .. settings.flingTarget, yPos, function()
            local targets = {"All", "Murderer", "Sheriff", "Innocent"}
            local idx = table.find(targets, settings.flingTarget) or 1
            settings.flingTarget = targets[idx % #targets + 1]
            btn.Text = "Цель: " .. settings.flingTarget
        end)
        yPos = yPos + 0.1
        createSlider("Сила", yPos, 100, 10000, settings.flingPower, function(v) settings.flingPower = v end)
        yPos = yPos + 0.1
        createSlider("Задержка", yPos, 0.05, 1, settings.flingDelay, function(v) settings.flingDelay = v end)
        yPos = yPos + 0.1
        createToggle("Вверх", yPos, true, function(s) settings.flingUp = s end)

    elseif funcName == "Ban" then
        createBtn("Тип: " .. settings.banType, yPos, function()
            local types = {"Kick", "Ban", "ESC", "Crash"}
            local idx = table.find(types, settings.banType) or 1
            settings.banType = types[idx % #types + 1]
            btn.Text = "Тип: " .. settings.banType
        end)
        yPos = yPos + 0.1
        createToggle("Массовый бан", yPos, false, function(s) settings.banMassive = s end)
        yPos = yPos + 0.1
        createBtn("Забанить всех", yPos, function()
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player then
                    pcall(function()
                        if settings.banType == "Ban" then
                            plr:Kick("🚫 Вы забанены!")
                        elseif settings.banType == "Kick" then
                            plr:Kick("👋 Вы кикнуты!")
                        elseif settings.banType == "ESC" then
                            game:GetService("VirtualInputManager"):SendKeyEvent(true, Enum.KeyCode.Escape, false, game)
                            task.wait(0.1)
                            game:GetService("VirtualInputManager"):SendKeyEvent(false, Enum.KeyCode.Escape, false, game)
                        elseif settings.banType == "Crash" then
                            -- Имитация краша
                            for i = 1, 100 do
                                spawn(function() plr:Kick("Краш") end)
                            end
                        end
                    end)
                end
            end
            miniMenu:Destroy()
        end)
        yPos = yPos + 0.1
        -- Список игроков для бана
        local list = Instance.new("ScrollingFrame")
        list.Size = UDim2.new(1,0,1,-yPos)
        list.Position = UDim2.new(0,0,yPos,0)
        list.BackgroundTransparency = 1
        list.CanvasSize = UDim2.new(0,0,0,0)
        list.ScrollBarThickness = 5
        list.Parent = frame

        local y = 0
        for _, plr in ipairs(game.Players:GetPlayers()) do
            if plr == player then continue end
            local btn = Instance.new("TextButton")
            btn.Size = UDim2.new(1,0,0,25)
            btn.Position = UDim2.new(0,0,0,y)
            btn.BackgroundColor3 = Color3.fromRGB(30,0,0)
            btn.BorderSizePixel = 0
            btn.Text = plr.Name
            btn.TextColor3 = Color3.fromRGB(255,255,255)
            btn.TextScaled = true
            btn.Font = Enum.Font.Gotham
            btn.Parent = list
            btn.MouseButton1Click:Connect(function()
                pcall(function()
                    if settings.banType == "Ban" then
                        plr:Kick("🚫 Вы забанены!")
                    elseif settings.banType == "Kick" then
                        plr:Kick("👋 Вы кикнуты!")
                    elseif settings.banType == "ESC" then
                        game:GetService("VirtualInputManager"):SendKeyEvent(true, Enum.KeyCode.Escape, false, game)
                        task.wait(0.1)
                        game:GetService("VirtualInputManager"):SendKeyEvent(false, Enum.KeyCode.Escape, false, game)
                    elseif settings.banType == "Crash" then
                        for i = 1, 100 do
                            spawn(function() plr:Kick("Краш") end)
                        end
                    end
                end)
                miniMenu:Destroy()
            end)
            y = y + 30
        end
        list.CanvasSize = UDim2.new(0,0,0,y)

    elseif funcName == "Kill All" then
        createSlider("Задержка", yPos, 0.5, 10, settings.killAllDelay, function(v) settings.killAllDelay = v end)
        yPos = yPos + 0.1
        createToggle("Только убийца", yPos, false, function(s) settings.killAllOnlyMurderer = s end)
        yPos = yPos + 0.1
        createBtn("Убить всех", yPos, function()
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character and plr.Character:FindFirstChild("Humanoid") then
                    if settings.killAllOnlyMurderer then
                        if getRole(plr) ~= "Murderer" then continue end
                    end
                    plr.Character.Humanoid.Health = 0
                end
            end
            miniMenu:Destroy()
        end)
    end
end
