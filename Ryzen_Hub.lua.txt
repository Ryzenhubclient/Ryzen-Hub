-- ===== ЗАГРУЗКА/СОХРАНЕНИЕ КОНФИГА =====
local configName = "KabancikHubConfig"

local function loadConfig()
    local success, data = pcall(function()
        return httpService:JSONDecode(httpService:GetAsync(configName))
    end)
    if success and data then
        for key, value in pairs(data) do
            if settings[key] ~= nil then
                settings[key] = value
            end
        end
        print("✅ Конфиг загружен!")
        return true
    end
    print("⚠️ Конфиг не найден, создан новый.")
    return false
end

local function saveConfig()
    local data = {}
    for key, value in pairs(settings) do
        data[key] = value
    end
    local success, err = pcall(function()
        httpService:SetAsync(configName, httpService:JSONEncode(data))
    end)
    if success then
        print("✅ Конфиг сохранён!")
    else
        warn("❌ Ошибка сохранения: " .. tostring(err))
    end
end

-- ===== INVISIBLE (НЕВИДИМОСТЬ) =====
local function applyInvisible(state)
    local character = player.Character
    if not character then return end
    
    local transparency = state and settings.invisibleTransparency or 0
    
    -- Все части тела
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") or part:IsA("MeshPart") or part:IsA("Part") then
            part.Transparency = transparency
        end
    end
    
    -- Аксессуары (шляпы, рюкзаки и т.д.)
    for _, accessory in ipairs(character:GetChildren()) do
        if accessory:IsA("Accessory") then
            for _, part in ipairs(accessory:GetDescendants()) do
                if part:IsA("BasePart") or part:IsA("MeshPart") or part:IsA("Part") then
                    part.Transparency = transparency
                end
            end
        end
    end
end

local function toggleInvisible(state)
    toggles.invisible = state
    applyInvisible(state)
    print(state and "👻 Невидимость ВКЛЮЧЕНА" or "👻 Невидимость ВЫКЛЮЧЕНА")
end

-- Отслеживание новых частей тела (для персонажа после респавна)
local function watchForNewParts()
    local character = player.Character
    if not character then return end
    
    character.ChildAdded:Connect(function(child)
        if toggles.invisible then
            if child:IsA("BasePart") or child:IsA("MeshPart") or child:IsA("Part") then
                child.Transparency = settings.invisibleTransparency
            end
            if child:IsA("Accessory") then
                for _, part in ipairs(child:GetDescendants()) do
                    if part:IsA("BasePart") or part:IsA("MeshPart") or part:IsA("Part") then
                        part.Transparency = settings.invisibleTransparency
                    end
                end
            end
        end
    end)
end

-- ===== ОПРЕДЕЛЕНИЕ РОЛИ =====
local function getRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    for _, tool in ipairs(plr.Character:GetChildren()) do
        if tool:IsA("Tool") then
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    if plr.Backpack then
        for _, tool in ipairs(plr.Backpack:GetChildren()) do
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    return "Innocent"
end

-- ===== ПОЛУЧЕНИЕ ЧАСТИ ТЕЛА =====
local function getTargetPart(character)
    if not character then return nil end
    local partName = "Head"
    if settings.silentAimPart == "Живот" then
        partName = "Torso"
    elseif settings.silentAimPart == "Ноги" then
        partName = "HumanoidRootPart"
    end
    return character:FindFirstChild(partName)
end

-- ===== SILENT AIM =====
local function startSilentAim()
    if silentAimConnection then
        silentAimConnection:Disconnect()
        silentAimConnection = nil
    end
    if not toggles.silentAim then return end

    silentAimConnection = rs.Heartbeat:Connect(function()
        if not player.Character then return end
        local character = player.Character
        local root = character:FindFirstChild("HumanoidRootPart")
        if not root then return end

        local target = nil
        local targetDist = settings.silentAimRange
        
        for _, plr in ipairs(game.Players:GetPlayers()) do
            if plr ~= player and plr.Character then
                local role = getRole(plr)
                if settings.silentAimTarget == "All" or settings.silentAimTarget == role then
                    local targetRoot = plr.Character:FindFirstChild("HumanoidRootPart")
                    if targetRoot then
                        local dist = (targetRoot.Position - root.Position).Magnitude
                        if dist < targetDist then
                            targetDist = dist
                            target = plr
                        end
                    end
                end
            end
        end

        if target then
            silentAimTarget = target
            local targetPart = getTargetPart(target.Character)
            if targetPart then
                local direction = (targetPart.Position - root.Position).Unit
                local lookAt = CFrame.lookAt(root.Position, root.Position + direction)
                root.CFrame = lookAt
                
                if settings.silentAimLock then
                    local screenPos, onScreen = workspace.CurrentCamera:WorldToViewportPoint(targetPart.Position)
                    if onScreen then
                        mouse.Move(screenPos.X, screenPos.Y)
                    end
                end
                
                if settings.silentAimAutoShoot then
                    local tool = character:FindFirstChildOfClass("Tool")
                    if tool then
                        mouse1click()
                        task.wait(0.1)
                    end
                end
            end
        else
            silentAimTarget = nil
        end
    end)
end

-- ===== ESP =====
function updateESP()
    for _, v in pairs(espObjects) do v:Destroy() end
    espObjects = {}
    if not toggles.esp then return end

    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local role = getRole(plr)
            local color = Color3.fromRGB(0, 255, 0)
            if role == "Murderer" then color = Color3.fromRGB(255, 0, 0)
            elseif role == "Sheriff" then color = Color3.fromRGB(0, 100, 255) end

            local esp = Instance.new("BillboardGui")
            esp.Size = UDim2.new(0, 200, 0, 50)
            esp.AlwaysOnTop = true
            esp.Parent = plr.Character.HumanoidRootPart

            local label = Instance.new("TextLabel")
            label.Size = UDim2.new(1, 0, 1, 0)
            label.BackgroundTransparency = 1
            label.TextColor3 = color
            label.TextScaled = true
            label.Font = Enum.Font.GothamBold
            
            if settings.espMode == "Имена" then
                label.Text = plr.Name
            elseif settings.espMode == "Роли" then
                label.Text = role
            else
                local dist = (plr.Character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
                label.Text = math.floor(dist) .. "м"
            end
            label.Parent = esp

            table.insert(espObjects, esp)
        end
    end
end

-- ===== ПРИМЕНЕНИЕ ХИТБОКСОВ =====
local function applyHitboxes()
    if not toggles.hitbox then return end
    
    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character then
            for _, part in ipairs(plr.Character:GetChildren()) do
                if part:IsA("BasePart") then
                    local name = part.Name:lower()
                    local size = Vector3.new(2, 2, 1)
                    
                    if name:find("head") then
                        size = Vector3.new(settings.hitboxHead, settings.hitboxHead, settings.hitboxHead)
                    elseif name:find("torso") or name:find("chest") or name:find("humanoidrootpart") then
                        size = Vector3.new(settings.hitboxTorso, settings.hitboxTorso, settings.hitboxTorso)
                    elseif name:find("leftarm") or name:find("left arm") then
                        size = Vector3.new(settings.hitboxLeftArm, settings.hitboxLeftArm, settings.hitboxLeftArm)
                    elseif name:find("rightarm") or name:find("right arm") then
                        size = Vector3.new(settings.hitboxRightArm, settings.hitboxRightArm, settings.hitboxRightArm)
                    elseif name:find("leftleg") or name:find("left leg") then
                        size = Vector3.new(settings.hitboxLeftLeg, settings.hitboxLeftLeg, settings.hitboxLeftLeg)
                    elseif name:find("rightleg") or name:find("right leg") then
                        size = Vector3.new(settings.hitboxRightLeg, settings.hitboxRightLeg, settings.hitboxRightLeg)
                    end
                    
                    part.Size = size
                end
            end
        end
    end
end
