-- ===== ОПРЕДЕЛЕНИЕ РОЛИ =====
local function getRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    for _, tool in ipairs(plr.Character:GetChildren()) do
        if tool:IsA("Tool") then
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    if plr.Backpack then
        for _, tool in ipairs(plr.Backpack:GetChildren()) do
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    return "Innocent"
end

-- ===== ESP ЧЕРЕЗ СТЕНЫ =====
function updateESP()
    for _, v in pairs(espObjects) do v:Destroy() end
    espObjects = {}
    if not toggles.esp then return end
    
    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local role = getRole(plr)
            local color = settings.espColorInnocent
            if role == "Sheriff" then color = settings.espColorSheriff
            elseif role == "Murderer" then color = settings.espColorMurderer end
            
            local esp = Instance.new("BillboardGui")
            esp.Size = UDim2.new(0, 200, 0, 60)
            esp.AlwaysOnTop = true
            esp.Parent = plr.Character.HumanoidRootPart
            
            local frame = Instance.new("Frame")
            frame.Size = UDim2.new(1, 0, 1, 0)
            frame.BackgroundTransparency = 1
            frame.Parent = esp
            
            local label = Instance.new("TextLabel")
            label.Size = UDim2.new(1, 0, 0.5, 0)
            label.Position = UDim2.new(0, 0, 0, 0)
            label.BackgroundTransparency = 1
            label.TextColor3 = color
            label.TextScaled = true
            label.Font = Enum.Font.GothamBold
            
            if settings.espMode == "Имена" then
                label.Text = plr.Name
            elseif settings.espMode == "Роли" then
                label.Text = role
            else
                local dist = (plr.Character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
                label.Text = math.floor(dist) .. "м"
            end
            label.Parent = frame
            
            local roleLabel = Instance.new("TextLabel")
            roleLabel.Size = UDim2.new(1, 0, 0.5, 0)
            roleLabel.Position = UDim2.new(0, 0, 0.5, 0)
            roleLabel.BackgroundTransparency = 1
            roleLabel.TextColor3 = color
            roleLabel.TextScaled = true
            roleLabel.Font = Enum.Font.Gotham
            roleLabel.Text = "[" .. role .. "]"
            roleLabel.Parent = frame
            
            -- Показ здоровья
            if settings.espShowHealth then
                local health = Instance.new("TextLabel")
                health.Size = UDim2.new(1, 0, 0.3, 0)
                health.Position = UDim2.new(0, 0, 0.7, 0)
                health.BackgroundTransparency = 1
                health.TextColor3 = Color3.fromRGB(255, 255, 255)
                health.TextScaled = true
                health.Font = Enum.Font.Gotham
                local hp = plr.Character:FindFirstChild("Humanoid")
                health.Text = "❤️ " .. (hp and math.floor(hp.Health) or "?")
                health.Parent = frame
            end
            
            -- Показ расстояния
            if settings.espShowDistance then
                local dist = Instance.new("TextLabel")
                dist.Size = UDim2.new(1, 0, 0.3, 0)
                dist.Position = UDim2.new(0, 0, 0.7, 0)
                dist.BackgroundTransparency = 1
                dist.TextColor3 = Color3.fromRGB(255, 255, 255)
                dist.TextScaled = true
                dist.Font = Enum.Font.Gotham
                local distance = (plr.Character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
                dist.Text = "📏 " .. math.floor(distance) .. "м"
                dist.Parent = frame
            end
            
            table.insert(espObjects, esp)
        end
    end
end

-- ===== AUTO LOOTGUN =====
local function startLootgun()
    if lootgunConnection then
        lootgunConnection:Disconnect()
        lootgunConnection = nil
    end
    if not toggles.lootgun then return end
    
    lootgunConnection = rs.Heartbeat:Connect(function()
        if not player.Character then return end
        local character = player.Character
        local root = character:FindFirstChild("HumanoidRootPart")
        if not root then return end
        
        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("Tool") then
                local name = v.Name:lower()
                if name:find("gun") or name:find("pistol") or name:find("revolver") or name:find("sheriff") then
                    if v.Parent ~= character and v.Parent ~= player.Backpack then
                        local distance = 999
                        local handle = v:FindFirstChild("Handle")
                        if handle then
                            distance = (handle.Position - root.Position).Magnitude
                        end
                        if distance <= settings.lootgunRange then
                            pcall(function()
                                local hasGun = false
                                for _, tool in ipairs(character:GetChildren()) do
                                    if tool:IsA("Tool") then
                                        local tName = tool.Name:lower()
                                        if tName:find("gun") or tName:find("pistol") or tName:find("revolver") then
                                            hasGun = true
                                            break
                                        end
                                    end
                                end
                                if not hasGun then
                                    if v.Parent and v.Parent:IsA("Model") and v.Parent:FindFirstChild("Humanoid") then
                                        return
                                    end
                                    v.Parent = character
                                    print("🔫 Пистолет подобран!")
                                end
                            end)
                        end
                    end
                end
            end
        end
    end)
end

            settings.banType = types[idx % #types + 1]
            btn.Text = "Тип: " .. settings.banType
        end)
        
    -- НАСТРОЙКИ LOOTGUN
    elseif funcName == "Lootgun" then
        createSlider("Радиус", yPos, 10, 100, settings.lootgunRange, function(v) settings.lootgunRange = v end)
        yPos = yPos + 0.1
        createBtn("Проверить", yPos, function()
            for _, v in ipairs(workspace:GetDescendants()) do
                if v:IsA("Tool") and (v.Name:lower():find("gun") or v.Name:lower():find("pistol")) then
                    if v.Parent ~= player.Character and v.Parent ~= player.Backpack then
                        pcall(function() v.Parent = player.Character print("🔫 Пистолет подобран!") end)
                    end
                end
            end
        end)
        
    -- НАСТРОЙКИ ВИЗУАЛОВ
    elseif funcName == "Аниме Небо" then
        createBtn("Облака: " .. (toggles.animeSky and "Вкл" or "Выкл"), yPos, function()
            toggles.animeSky = not toggles.animeSky
            btn.Text = "Облака: " .. (toggles.animeSky and "Вкл" or "Выкл")
            updateVisuals()
        end)
        
    elseif funcName == "Туман" then
        createSlider("Дальность", yPos, 10, 500, settings.fogEnd, function(v) 
            settings.fogEnd = v 
            createFog()
        end)
        yPos = yPos + 0.1
        createBtn("Цвет: Фиолетовый", yPos, function()
            local colors = {"Фиолетовый", "Розовый", "Голубой", "Белый"}
            local idx = table.find(colors, settings.fogColorName) or 1
            settings.fogColorName = colors[idx % #colors + 1]
            btn.Text = "Цвет: " .. settings.fogColorName
            if settings.fogColorName == "Фиолетовый" then settings.fogColor = Color3.fromRGB(200, 100, 255)
            elseif settings.fogColorName == "Розовый" then settings.fogColor = Color3.fromRGB(255, 150, 200)
            elseif settings.fogColorName == "Голубой" then settings.fogColor = Color3.fromRGB(150, 200, 255)
            else settings.fogColor = Color3.fromRGB(255, 255, 255) end
            createFog()
        end)
        
    elseif funcName == "Атмосфера" then
        createBtn("Цвет: Фиолетовый", yPos, function()
            local colors = {"Фиолетовый", "Розовый", "Голубой", "Зеленый"}
            local idx = table.find(colors, settings.atmoColorName) or 1
            settings.atmoColorName = colors[idx % #colors + 1]
            btn.Text = "Цвет: " .. settings.atmoColorName
            if settings.atmoColorName == "Фиолетовый" then settings.atmosphereColor = Color3.fromRGB(150, 50, 255)
            elseif settings.atmoColorName == "Розовый" then settings.atmosphereColor = Color3.fromRGB(255, 100, 200)
            elseif settings.atmoColorName == "Голубой" then settings.atmosphereColor = Color3.fromRGB(100, 200, 255)
            else settings.atmosphereColor = Color3.fromRGB(100, 255, 150) end
            createAtmosphere()
        end)
        
    elseif funcName == "Bloom" then
        createSlider("Интенсивность", yPos, 0, 2, 0.5, function(v) 
            if visualEffects.bloom then visualEffects.bloom.Intensity = v / 2 end
        end)
        yPos = yPos + 0.1
        createSlider("Размер", yPos, 10, 200, 100, function(v) 
            if visualEffects.bloom then visualEffects.bloom.Size = v end
        end)
        
    elseif funcName == "Цветокоррекция" then
        createSlider("Яркость", yPos, -1, 1, 0.1, function(v) 
            if visualEffects.colorCorrection then visualEffects.colorCorrection.Brightness = v end
        end)
        yPos = yPos + 0.1
        createSlider("Насыщенность", yPos, -1, 1, 0.2, function(v) 
            if visualEffects.colorCorrection then visualEffects.colorCorrection.Saturation = v end
        end)
    end
end
