-- ===== ОПРЕДЕЛЕНИЕ РОЛИ =====
local function getRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    for _, tool in ipairs(plr.Character:GetChildren()) do
        if tool:IsA("Tool") then
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    if plr.Backpack then
        for _, tool in ipairs(plr.Backpack:GetChildren()) do
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    return "Innocent"
end

local function getPlayerAvatar(player)
    local userId = player.UserId
    local avatar = Instance.new("ImageLabel")

    avatar.BackgroundTransparency = 1
    avatar.Image = "https://www.roblox.com/headshot-thumbnail/image?userId=" .. userId .. "&width=420&height=420&format=png"
    return avatar
end

-- ===== FLING PLAYER (ВЫКИДЫВАЕТ ЗА КАРТУ) =====
local function flingPlayerByRightClick()
    if not toggles.flingPlayer then return end
    if not mouse.Target then return end
    local targetPart = mouse.Target
    if not targetPart.Parent then return end
    local targetCharacter = targetPart.Parent
    if not targetCharacter:FindFirstChild("Humanoid") then return end
    local targetPlayer = game.Players:GetPlayerFromCharacter(targetCharacter)
    if not targetPlayer or targetPlayer == player then return end
    
    -- Проверяем, подходит ли цель по настройке
    local role = getRole(targetPlayer)
    if settings.flingTarget ~= "All" and settings.flingTarget ~= role then return end
    
    pcall(function()
        if targetCharacter:FindFirstChild("HumanoidRootPart") then
            -- Выкидываем за карту
            targetCharacter.HumanoidRootPart.CFrame = CFrame.new(0, -500, 0)
        end
    end)
end

mouse.Button2Down:Connect(function()
    flingPlayerByRightClick()
end)
