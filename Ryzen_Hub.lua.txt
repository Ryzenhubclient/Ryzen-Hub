-- ===== ФУНКЦИЯ ОПРЕДЕЛЕНИЯ РОЛИ =====
local function getPlayerRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    local hasKnife, hasGun = false, false
    for _, tool in ipairs(plr.Character:GetChildren()) do
        if tool:IsA("Tool") then
            local name = tool.Name:lower()
            if name:find("knife") or name:find("dagger") then hasKnife = true end
            if name:find("gun") or name:find("pistol") or name:find("revolver") then hasGun = true end
        end
    end
    if plr.Backpack then
        for _, tool in ipairs(plr.Backpack:GetChildren()) do
            if tool:IsA("Tool") then
                local name = tool.Name:lower()
                if name:find("knife") or name:find("dagger") then hasKnife = true end
                if name:find("gun") or name:find("pistol") or name:find("revolver") then hasGun = true end
            end
        end
    end
    if hasKnife and hasGun then return "Sheriff"
    elseif hasKnife then return "Murderer"
    elseif hasGun then return "Sheriff"
    else return "Innocent" end
end

-- ===== ESP =====
local function updateESP()
    for _, obj in ipairs(espObjects) do
        if obj and obj.Parent then obj:Destroy() end
    end
    espObjects = {}
    if not toggles.esp then return end

    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr == player or not plr.Character or not plr.Character:FindFirstChild("Head") then continue end
        local role = getPlayerRole(plr)
        local show = false
        if role == "Murderer" and toggles.showMurderer then show = true end
        if role == "Sheriff" and toggles.showSheriff then show = true end
        if role == "Innocent" and toggles.showInnocent then show = true end
        if not show then continue end

        local color = role == "Murderer" and Color3.fromRGB(255, 0, 0) or 
                     role == "Sheriff" and Color3.fromRGB(0, 100, 255) or 
                     Color3.fromRGB(0, 255, 0)

        local esp = Instance.new("BillboardGui")
        esp.Name = "ESP_Ryzen"
        esp.Size = UDim2.new(0, 150, 0, 40)
        esp.AlwaysOnTop = true
        esp.Parent = plr.Character.Head
        table.insert(espObjects, esp)

        local text = Instance.new("TextLabel")
        text.Size = UDim2.new(1, 0, 1, 0)
        text.BackgroundTransparency = 1
        text.Text = plr.Name .. " [" .. role .. "]"
        text.TextColor3 = color
        text.TextScaled = true
        text.Font = Enum.Font.GothamBold
        text.Parent = esp
        table.insert(espObjects, text)

        local highlight = Instance.new("Highlight")
        highlight.Name = "ESP_Highlight"
        highlight.Adornee = plr.Character
        highlight.FillColor = color
        highlight.FillTransparency = 0.5
        highlight.OutlineColor = color
        highlight.OutlineTransparency = 0.2
        highlight.Parent = plr.Character
        table.insert(espObjects, highlight)
    end
end

-- ===== TRIGGER BOT =====
local function toggleTriggerBot(state)
    toggles.triggerbot = state
    if state then
        if triggerbotConnection then triggerbotConnection:Disconnect() end
        triggerbotConnection = game:GetService("RunService").Heartbeat:Connect(function()
            if not toggles.triggerbot then 
                if triggerbotConnection then triggerbotConnection:Disconnect() end
                return 
            end
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character and plr.Character:FindFirstChild("Humanoid") then
                    if getPlayerRole(plr) == "Murderer" then
                        plr.Character.Humanoid.Health = 0
                        print("🔫 Мардер убит (Trigger Bot)")
                    end
                end
            end
        end)
    else
        if triggerbotConnection then triggerbotConnection:Disconnect() end
    end
end

-- ===== SHOT MURDERER =====
local function toggleShotMurderer(state)
    toggles.shotMurderer = state
    if state then

    else

local function killMurderer()
    if not toggles.shotMurderer then return end
    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("Humanoid") then
            if getPlayerRole(plr) == "Murderer" then
                plr.Character.Humanoid.Health = 0
                print("🔫 Мардер убит!")
                return
            end
        end
    end
    print("❌ Мардер не найден!")
end

-- ===== BUNNY HOP =====
local function toggleBunnyHop(state)
    toggles.bunnyhop = state
    if state then
        if bunnyHopConnection then bunnyHopConnection:Disconnect() end
        bunnyHopConnection = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.bunnyhop then return end
            if player.Character and player.Character:FindFirstChild("Humanoid") then
                local humanoid = player.Character.Humanoid
                if humanoid:GetState() == Enum.HumanoidStateType.Jumping then
                    humanoid.WalkSpeed = settings.bunnyhop
                else
                    humanoid.WalkSpeed = 16
                end
            end
        end)
    else
        if bunnyHopConnection then bunnyHopConnection:Disconnect() end
        if player.Character and player.Character:FindFirstChild("Humanoid") then
            player.Character.Humanoid.WalkSpeed = 16
        end
    end
end

-- ===== SPIN BOT =====
local function toggleSpinBot(state)

-- ===== АВТО-ПОДБОР ПИСТОЛЕТА ПОСЛЕ СМЕРТИ =====
local function autoPickupGun()
    if not settings.autoPickupGun then return end
    task.wait(1) -- ждём возрождения
    if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
        local character = player.Character
        local root = character.HumanoidRootPart
        local gunFound = false
        for _, item in ipairs(workspace:GetDescendants()) do
            if item:IsA("Tool") then
                local name = item.Name:lower()
                if name:find("gun") or name:find("pistol") or name:find("revolver") then
                    if item.Parent ~= character and item.Parent ~= player.Backpack then
                        local dist = (item.Position - root.Position).Magnitude
                        if dist < 100 then -- ищем в радиусе 100
                            -- телепортируем к пистолету и забираем
                            root.CFrame = CFrame.new(item.Position + Vector3.new(0, 2, 0))
                            task.wait(0.2)
                            item.Parent = character
                            print("🔫 Пистолет автоматически подобран после смерти!")
                            gunFound = true
                            break
                        end
                    end
                end
            end
        end
        if not gunFound then
            print("❌ Пистолет не найден на карте.")
        end
    end
end

-- Подключаем авто-подбор к событию смерти/возрождения
player.CharacterAdded:Connect(autoPickupGun)

-- ===== ЗАБРАТЬ ПИСТОЛЕТ С КАРТЫ ПО НАЖАТИЮ КНОПКИ =====
local function pickupGunFromMap()
    if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
    local root = player.Character.HumanoidRootPart
    local gunFound = false
    for _, item in ipairs(workspace:GetDescendants()) do
        if item:IsA("Tool") then
            local name = item.Name:lower()
            if name:find("gun") or name:find("pistol") or name:find("revolver") then
                if item.Parent ~= player.Character and item.Parent ~= player.Backpack then
                    local dist = (item.Position - root.Position).Magnitude
                    if dist < 100 then
                        root.CFrame = CFrame.new(item.Position + Vector3.new(0, 2, 0))
                        task.wait(0.2)
                        item.Parent = player.Character
                        print("🔫 Пистолет подобран с карты!")
                        gunFound = true
                        break
                    end
                end
            end
        end
    end
    if not gunFound then
        print("❌ Пистолет не найден на карте (или слишком далеко).")
    end
end

-- Клавиша G для забора пистолета
game:GetService("UserInputService").InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.G and not input.UserInputType == Enum.UserInputType.Keyboard then return end
    if input.KeyCode == Enum.KeyCode.G then
        pickupGunFromMap()
    end
end)

-- ===== ОСТАЛЬНЫЕ ФУНКЦИИ =====
local function toggleJump(state)
    toggles.jump = state
    if player.Character and player.Character:FindFirstChild("Humanoid") then
        player.Character.Humanoid.JumpPower = state and settings.jumppower or 50
    end
end

local function enableNoclip()
    if noclipConnection then noclipConnection:Disconnect() end
    noclipActive = true
    noclipStartTime = tick()
    noclipConnection = game:GetService("RunService").Stepped:Connect(function()
        if not noclipActive then return end
        if tick() - noclipStartTime >= 86400 then
            noclipActive = false
            if noclipConnection then noclipConnection:Disconnect() end
            print("⏰ Noclip отключён")
            return
        end
        if player.Character then
            for _, part in ipairs(player.Character:GetChildren()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end
        end
    end)
end

local function disableNoclip()
    noclipActive = false
    if noclipConnection then noclipConnection:Disconnect() end
    if player.Character then
        for _, part in ipairs(player.Character:GetChildren()) do
            if part:IsA("BasePart") then part.CanCollide = true end
        end
    end
end

local function toggleNoclip(state)
    if state then enableNoclip() else disableNoclip() end
    toggles.noclip = state
end

local function toggleTeleport(state)
    toggles.teleport = state
    if state then
        mouse.Button1Down:Connect(function()
            if toggles.teleport and mouse.Target then
                local targetPos = mouse.Target.Position
                if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    player.Character.HumanoidRootPart.CFrame = CFrame.new(targetPos)
                end
            end
        end)
    end
end

local function toggleFly(state)
    toggles.fly = state
    if state then
        if flyConnection then flyConnection:Disconnect() end
        flyConnection = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.fly then 
                if flyConnection then flyConnection:Disconnect() end
                return 
            end
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                local root = player.Character.HumanoidRootPart
                local moveDir = Vector3.new(0, 0, 0)
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + root.CFrame.LookVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - root.CFrame.LookVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - root.CFrame.RightVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + root.CFrame.RightVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end
                root.Velocity = moveDir * settings.flyspeed
            end
        end)
    else
        if flyConnection then flyConnection:Disconnect() end
        if player.Character and player.Character:FindFirstChild("Humanoid") then
            player.Character.Humanoid.PlatformStand = false
        end
    end
end

local function toggleFps(state)
    toggles.showFps = state
    if state then
        if fpsLabel then fpsLabel:Destroy() end
        local screenGui = Instance.new("ScreenGui")
        screenGui.Name = "FPSDisplay"
        screenGui.Parent = game.CoreGui
        fpsLabel = Instance.new("TextLabel")
        fpsLabel.Size = UDim2.new(0, 100, 0, 30)
        fpsLabel.Position = UDim2.new(0, 10, 0, 10)
        fpsLabel.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
        fpsLabel.BackgroundTransparency = 0.3
        fpsLabel.BorderSizePixel = 0
        fpsLabel.Text = "FPS: 0"
        fpsLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
        fpsLabel.TextScaled = true
        fpsLabel.Font = Enum.Font.GothamBold
        fpsLabel.Parent = screenGui
        local frames = 0
        local lastTime = tick()
        game:GetService("RunService").Heartbeat:Connect(function()
            frames = frames + 1
            if tick() - lastTime >= 1 then
                if fpsLabel then fpsLabel.Text = "FPS: " .. frames end
                frames = 0
                lastTime = tick()
            end
        end)
    else
        if fpsLabel then fpsLabel:Destroy() end
    end
end

-- ===== ФУНКЦИЯ СМЕНЫ ТЕМЫ =====
local function applyTheme(themeName)
    currentTheme = themeName
    local theme = themes[themeName]
    if not theme then return end
    if gui then
        local mainFrame = gui:FindFirstChild("MainFrame")
        if mainFrame then
            mainFrame.BackgroundColor3 = theme.bg
            mainFrame.BorderColor3 = theme.border
        end
        local titleFrame = gui:FindFirstChild("MainFrame"):FindFirstChild("TitleFrame")
        if titleFrame then
            titleFrame.BackgroundColor3 = theme.bg + Color3.fromRGB(20,0,0) -- чуть светлее
        end
        local title = titleFrame and titleFrame:FindFirstChild("Title")
        if title then
            title.TextColor3 = theme.accent
        end
        for _, btn in ipairs(gui:GetDescendants()) do
            if btn:IsA("TextButton") and btn.Name ~= "CloseBtn" then
                if btn.BackgroundColor3 ~= Color3.fromRGB(150,0,0) then -- не трогаем кнопку убийства
                    btn.BackgroundColor3 = theme.bg + Color3.fromRGB(30,0,0)
                    btn.BorderColor3 = theme.border
                    btn.TextColor3 = theme.text
                end
            end
        end
        print("🎨 Тема изменена на " .. themeName)
    end
end

    -- НОВЫЙ РАЗДЕЛ: НАСТРОЙКИ GUI И ПИСТОЛЕТ
    createSection("SETTINGS", 1.00)
    createButton("🎨 Сменить тему", 1.05, function()
        local themeNames = {"Red","Blue","Green","Purple","Gold"}
        local currentIndex = table.find(themeNames, currentTheme) or 1
        local nextIndex = currentIndex % #themeNames + 1
        applyTheme(themeNames[nextIndex])
    end)
    createButton("🔫 Авто-подбор после смерти", 1.10, function(state)
        settings.autoPickupGun = state
        print(state and "✅ Авто-подбор включён" or "❌ Авто-подбор выключен")
    end)
    createButton("🔫 Забрать пистолет [G]", 1.15, function()
        pickupGunFromMap()
    end)

    -- СЛАЙДЕРЫ (подняты выше)
    createSlider("🐇 Bunny Speed", 1.20, 20, 300, settings.bunnyhop, function(val)
        settings.bunnyhop = val
    end)
    createSlider("✈️ Fly Speed", 1.26, 10, 200, settings.flyspeed, function(val)
        settings.flyspeed = val
    end)
    createSlider("🦘 Jump Power", 1.32, 50, 300, settings.jumppower, function(val)
        settings.jumppower = val
    end)

    print("✅ Ryzen Hub загружен!")
end

-- ===== ОТКРЫТИЕ ПО ПРАВОМУ SHIFT =====
game:GetService("UserInputService").InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.RightShift then
        if gui and gui.Parent then
            gui:Destroy()
            gui = nil
        else
            createRyzenHub()
        end
    end
end)

-- Обработчик клавиши H (убийство мардера)
game:GetService("UserInputService").InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.H then
        killMurderer()
    end
end)
