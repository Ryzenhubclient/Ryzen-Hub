-- RYZEN HUB - ИСПРАВЛЕННАЯ ВЕРСИЯ
local player = game.Players.LocalPlayer
local mouse = player:GetMouse()

-- ===== НАСТРОЙКИ =====
local settings = {
    bunnyhop = 100,
    flyspeed = 50,
    jumppower = 120,
    spinSpeed = 10,
    autoPickupGun = false,
}

local toggles = {
    esp = false,
    showMurderer = true,
    showSheriff = true,
    showInnocent = false,
    triggerbot = false,
    bunnyhop = false,
    jump = false,
    noclip = false,
    teleport = false,
    fly = false,
    showFps = false,
    spinBot = false,
    fling = false,
    shotMurderer = false,
}

local espObjects = {}
local fpsLabel = nil
local gui = nil

local noclipActive, noclipStartTime, noclipConnection = false, 0, nil
local triggerbotConnection, bunnyHopConnection, spinConnection, flingConnection, flyConnection = nil, nil, nil, nil, nil

-- ===== ОПРЕДЕЛЕНИЕ РОЛИ (УЛУЧШЕНО) =====
local function getPlayerRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    for _, tool in ipairs(plr.Character:GetChildren()) do
        if tool:IsA("Tool") then
            local name = tool.Name:lower()
            if name:find("knife") or name:find("dagger") then return "Murderer" end
            if name:find("gun") or name:find("pistol") or name:find("revolver") then return "Sheriff" end
        end
    end
    if plr.Backpack then
        for _, tool in ipairs(plr.Backpack:GetChildren()) do
            if tool:IsA("Tool") then
                local name = tool.Name:lower()
                if name:find("knife") or name:find("dagger") then return "Murderer" end
                if name:find("gun") or name:find("pistol") or name:find("revolver") then return "Sheriff" end
            end
        end
    end
    return "Innocent"
end

-- ===== ESP =====
local function updateESP()
    for _, obj in ipairs(espObjects) do
        pcall(function() obj:Destroy() end)
    end
    espObjects = {}
    if not toggles.esp then return end

    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr == player then continue end
        if not plr.Character or not plr.Character:FindFirstChild("Head") then continue end
        
        local role = getPlayerRole(plr)
        local show = false
        if role == "Murderer" and toggles.showMurderer then show = true end
        if role == "Sheriff" and toggles.showSheriff then show = true end
        if role == "Innocent" and toggles.showInnocent then show = true end
        if not show then continue end

        local color = role == "Murderer" and Color3.fromRGB(255, 0, 0) or 
                     role == "Sheriff" and Color3.fromRGB(0, 100, 255) or 
                     Color3.fromRGB(0, 255, 0)

        local esp = Instance.new("BillboardGui")
        esp.Name = "ESP_Ryzen"
        esp.Size = UDim2.new(0, 150, 0, 40)
        esp.AlwaysOnTop = true
        esp.Parent = plr.Character.Head
        table.insert(espObjects, esp)

        local text = Instance.new("TextLabel")
        text.Size = UDim2.new(1, 0, 1, 0)
        text.BackgroundTransparency = 1
        text.Text = plr.Name .. " [" .. role .. "]"
        text.TextColor3 = color
        text.TextScaled = true
        text.Font = Enum.Font.GothamBold
        text.Parent = esp
        table.insert(espObjects, text)

        local highlight = Instance.new("Highlight")
        highlight.Name = "ESP_Highlight"
        highlight.Adornee = plr.Character
        highlight.FillColor = color
        highlight.FillTransparency = 0.5
        highlight.OutlineColor = color
        highlight.OutlineTransparency = 0.2
        highlight.Parent = plr.Character
        table.insert(espObjects, highlight)
    end
end

-- ===== TRIGGER BOT =====
local function toggleTriggerBot(state)
    toggles.triggerbot = state
    if state then
        if triggerbotConnection then triggerbotConnection:Disconnect() end
        triggerbotConnection = game:GetService("RunService").Heartbeat:Connect(function()
            if not toggles.triggerbot then 
                if triggerbotConnection then triggerbotConnection:Disconnect() end
                return 
            end
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character and plr.Character:FindFirstChild("Humanoid") then
                    if getPlayerRole(plr) == "Murderer" then
                        pcall(function()
                            plr.Character.Humanoid.Health = 0
                            print("🔫 Мардер убит (Trigger Bot)")
                        end)
                    end
                end
            end
        end)
    else
        if triggerbotConnection then triggerbotConnection:Disconnect() end
    end
end

-- ===== SHOT MURDERER =====
local function toggleShotMurderer(state)
    toggles.shotMurderer = state
    print(state and "🔫 Режим убийства мардера включён!" or "🔫 Режим убийства мардера выключен.")
end

local function killMurderer()
    if not toggles.shotMurderer then 
        print("❌ Режим убийства мардера выключен!")
        return 
    end
    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("Humanoid") then
            if getPlayerRole(plr) == "Murderer" then
                pcall(function()
                    plr.Character.Humanoid.Health = 0
                    print("🔫 Мардер убит!")
                end)
                return
            end
        end
    end
    print("❌ Мардер не найден!")
end

-- ===== BUNNY HOP =====
local function toggleBunnyHop(state)
    toggles.bunnyhop = state
    if state then
        if bunnyHopConnection then bunnyHopConnection:Disconnect() end
        bunnyHopConnection = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.bunnyhop then return end
            if player.Character and player.Character:FindFirstChild("Humanoid") then
                local humanoid = player.Character.Humanoid
                if humanoid:GetState() == Enum.HumanoidStateType.Jumping then
                    humanoid.WalkSpeed = settings.bunnyhop
                else
                    humanoid.WalkSpeed = 16
                end
            end
        end)
    else
        if bunnyHopConnection then bunnyHopConnection:Disconnect() end
        if player.Character and player.Character:FindFirstChild("Humanoid") then
            player.Character.Humanoid.WalkSpeed = 16
        end
    end
end

-- ===== SPIN BOT =====
local function toggleSpinBot(state)
    toggles.spinBot = state
    if state then
        if spinConnection then spinConnection:Disconnect() end
        spinConnection = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.spinBot then 
                if spinConnection then spinConnection:Disconnect() end
                return 
            end
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                player.Character.HumanoidRootPart.CFrame = player.Character.HumanoidRootPart.CFrame * CFrame.Angles(0, math.rad(settings.spinSpeed), 0)
            end
        end)
    else
        if spinConnection then spinConnection:Disconnect() end
    end
end

-- ===== FLING =====
local function toggleFling(state)
    toggles.fling = state
    if state then
        if flingConnection then flingConnection:Disconnect() end
        flingConnection = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.fling then 
                if flingConnection then flingConnection:Disconnect() end
                return 
            end
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                for _, plr in ipairs(game.Players:GetPlayers()) do
                    if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                        local dist = (plr.Character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
                        if dist < 10 then
                            pcall(function()
                                plr.Character.HumanoidRootPart.Velocity = Vector3.new(0, 1000, 0)
                            end)
                        end
                    end
                end
            end
        end)
    else
        if flingConnection then flingConnection:Disconnect() end
    end
end

-- ===== АВТО-ПОДБОР ПИСТОЛЕТА =====
local function autoPickupGun()
    if not settings.autoPickupGun then return end
    task.wait(1)
    if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
    
    local root = player.Character.HumanoidRootPart
    for _, item in ipairs(workspace:GetDescendants()) do
        if item:IsA("Tool") then
            local name = item.Name:lower()
            if name:find("gun") or name:find("pistol") or name:find("revolver") then
                if item.Parent ~= player.Character and item.Parent ~= player.Backpack then
                    local dist = (item.Position - root.Position).Magnitude
                    if dist < 100 then
                        pcall(function()
                            root.CFrame = CFrame.new(item.Position + Vector3.new(0, 2, 0))
                            task.wait(0.2)
                            item.Parent = player.Character
                            print("🔫 Пистолет автоматически подобран!")
                        end)
                        return
                    end
                end
            end
        end
    end
end

player.CharacterAdded:Connect(autoPickupGun)

-- ===== ЗАБРАТЬ ПИСТОЛЕТ ПО G =====
local function pickupGunFromMap()
    if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then 
        print("❌ Вы не в игре!")
        return 
    end
    local root = player.Character.HumanoidRootPart
    for _, item in ipairs(workspace:GetDescendants()) do
        if item:IsA("Tool") then
            local name = item.Name:lower()
            if name:find("gun") or name:find("pistol") or name:find("revolver") then
                if item.Parent ~= player.Character and item.Parent ~= player.Backpack then
                    local dist = (item.Position - root.Position).Magnitude
                    if dist < 100 then
                        pcall(function()
                            root.CFrame = CFrame.new(item.Position + Vector3.new(0, 2, 0))
                            task.wait(0.2)
                            item.Parent = player.Character
                            print("🔫 Пистолет подобран с карты!")
                        end)

                    end
                end
            end
        end
    end
    print("❌ Пистолет не найден на карте!")
end

-- ===== ОСТАЛЬНЫЕ ФУНКЦИИ =====
local function toggleJump(state)
    toggles.jump = state
    if player.Character and player.Character:FindFirstChild("Humanoid") then
        player.Character.Humanoid.JumpPower = state and settings.jumppower or 50
    end
end

local function enableNoclip()
    if noclipConnection then noclipConnection:Disconnect() end
    noclipActive = true
    noclipStartTime = tick()
    noclipConnection = game:GetService("RunService").Stepped:Connect(function()
        if not noclipActive then return end
        if tick() - noclipStartTime >= 86400 then
            noclipActive = false
            if noclipConnection then noclipConnection:Disconnect() end
            print("⏰ Noclip отключён")
            return
        end
        if player.Character then
            for _, part in ipairs(player.Character:GetChildren()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end
        end
    end)
end

local function disableNoclip()
    noclipActive = false
    if noclipConnection then noclipConnection:Disconnect() end
    if player.Character then
        for _, part in ipairs(player.Character:GetChildren()) do
            if part:IsA("BasePart") then part.CanCollide = true end
        end
    end
end

local function toggleNoclip(state)
    if state then enableNoclip() else disableNoclip() end
    toggles.noclip = state
end

local function toggleTeleport(state)
    toggles.teleport = state
    if state then
        mouse.Button1Down:Connect(function()
            if toggles.teleport and mouse.Target then
                local targetPos = mouse.Target.Position
                if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    player.Character.HumanoidRootPart.CFrame = CFrame.new(targetPos)
                end
            end
        end)
    end
end

local function toggleFly(state)
    toggles.fly = state
    if state then
        if flyConnection then flyConnection:Disconnect() end
        flyConnection = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.fly then 
                if flyConnection then flyConnection:Disconnect() end
                return 
            end
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                local root = player.Character.HumanoidRootPart
                local moveDir = Vector3.new(0, 0, 0)
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + root.CFrame.LookVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - root.CFrame.LookVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - root.CFrame.RightVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + root.CFrame.RightVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end
                root.Velocity = moveDir * settings.flyspeed
            end
        end)
    else
        if flyConnection then flyConnection:Disconnect() end
        if player.Character and player.Character:FindFirstChild("Humanoid") then
            player.Character.Humanoid.PlatformStand = false
        end
    end
end

local function toggleFps(state)
    toggles.showFps = state
    if state then
        if fpsLabel then fpsLabel:Destroy() end
        local screenGui = Instance.new("ScreenGui")
        screenGui.Name = "FPSDisplay"
        screenGui.Parent = game.CoreGui
        fpsLabel = Instance.new("TextLabel")
        fpsLabel.Size = UDim2.new(0, 100, 0, 30)
        fpsLabel.Position = UDim2.new(0, 10, 0, 10)
        fpsLabel.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
        fpsLabel.BackgroundTransparency = 0.3
        fpsLabel.BorderSizePixel = 0
        fpsLabel.Text = "FPS: 0"
        fpsLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
        fpsLabel.TextScaled = true
        fpsLabel.Font = Enum.Font.GothamBold
        fpsLabel.Parent = screenGui
        local frames = 0
        local lastTime = tick()
        game:GetService("RunService").Heartbeat:Connect(function()
            frames = frames + 1
            if tick() - lastTime >= 1 then
                if fpsLabel then fpsLabel.Text = "FPS: " .. frames end
                frames = 0
                lastTime = tick()
            end
        end)
    else
        if fpsLabel then fpsLabel:Destroy() end
    end
end

-- ===== ГЛАВНОЕ МЕНЮ =====
local function createRyzenHub()
    if gui then gui:Destroy() end

    gui = Instance.new("ScreenGui")
    gui.Name = "RyzenHubGUI"
    gui.Parent = game.CoreGui
    gui.ResetOnSpawn = false

    local mainFrame = Instance.new("Frame")
    mainFrame.Size = UDim2.new(0, 450, 0, 500)
    mainFrame.Position = UDim2.new(0.5, -225, 0.5, -250)
    mainFrame.BackgroundColor3 = Color3.fromRGB(20, 0, 0)
    mainFrame.BackgroundTransparency = 0.1
    mainFrame.BorderSizePixel = 2
    mainFrame.BorderColor3 = Color3.fromRGB(255, 0, 0)
    mainFrame.Active = true
    mainFrame.Draggable = true
    mainFrame.Parent = gui

    -- Заголовок
    local titleFrame = Instance.new("Frame")
    titleFrame.Size = UDim2.new(1, 0, 0, 45)
    titleFrame.BackgroundColor3 = Color3.fromRGB(40, 0, 0)
    titleFrame.BorderSizePixel = 0
    titleFrame.Parent = mainFrame

    local logo = Instance.new("ImageLabel")
    logo.Size = UDim2.new(0, 35, 0, 35)
    logo.Position = UDim2.new(0.02, 0, 0.1, 0)
    logo.BackgroundTransparency = 1
    logo.Image = "https://upload.wikimedia.org/wikipedia/commons/thumb/7/7f/AMD_Ryzen_logo.svg/120px-AMD_Ryzen_logo.svg.png"
    logo.Parent = titleFrame

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1, -50, 1, 0)
    title.Position = UDim2.new(0, 50, 0, 0)
    title.BackgroundTransparency = 1
    title.Text = "RYZEN HUB"
    title.TextColor3 = Color3.fromRGB(255, 50, 50)
    title.TextScaled = true
    title.Font = Enum.Font.GothamBold
    title.Parent = titleFrame

    local closeBtn = Instance.new("TextButton")
    closeBtn.Size = UDim2.new(0, 28, 0, 28)
    closeBtn.Position = UDim2.new(1, -36, 0, 8)
    closeBtn.BackgroundColor3 = Color3.fromRGB(100, 0, 0)
    closeBtn.BorderSizePixel = 0
    closeBtn.Text = "✕"
    closeBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    closeBtn.TextScaled = true
    closeBtn.Font = Enum.Font.GothamBold
    closeBtn.Parent = titleFrame
    closeBtn.MouseButton1Click:Connect(function() gui:Destroy() end)

    local function createButton(text, yPos, callback)
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(0.85, 0, 0, 28)
        btn.Position = UDim2.new(0.075, 0, yPos, 0)
        btn.BackgroundColor3 = Color3.fromRGB(50, 0, 0)
        btn.BorderSizePixel = 1
        btn.BorderColor3 = Color3.fromRGB(255, 50, 50)
        btn.Text = text
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        btn.TextScaled = true
        btn.Font = Enum.Font.Gotham
        btn.Parent = mainFrame

        local isActive = false
        btn.MouseButton1Click:Connect(function()
            isActive = not isActive
            if isActive then
                btn.BackgroundColor3 = Color3.fromRGB(100, 0, 0)
                btn.TextColor3 = Color3.fromRGB(255, 150, 150)
            else
                btn.BackgroundColor3 = Color3.fromRGB(50, 0, 0)
                btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            end
            callback(isActive)
            if text:find("ESP") then updateESP() end
        end)
        return btn
    end

    local function createSlider(text, yPos, minVal, maxVal, defaultVal, callback)
        local frame = Instance.new("Frame")
        frame.Size = UDim2.new(0.85, 0, 0, 25)
        frame.Position = UDim2.new(0.075, 0, yPos, 0)
        frame.BackgroundColor3 = Color3.fromRGB(40, 0, 0)
        frame.BorderSizePixel = 0
        frame.Parent = mainFrame

        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(0.4, 0, 1, 0)
        label.Position = UDim2.new(0.02, 0, 0, 0)
        label.BackgroundTransparency = 1
        label.Text = text
        label.TextColor3 = Color3.fromRGB(255, 200, 200)
        label.TextScaled = true
        label.Font = Enum.Font.Gotham
        label.Parent = frame

        local slider = Instance.new("TextButton")
        slider.Size = UDim2.new(0.4, 0, 0.7, 0)
        slider.Position = UDim2.new(0.45, 0, 0.15, 0)
        slider.BackgroundColor3 = Color3.fromRGB(60, 0, 0)
        slider.BorderSizePixel = 1
        slider.BorderColor3 = Color3.fromRGB(255, 50, 50)
        slider.Text = tostring(defaultVal)
        slider.TextColor3 = Color3.fromRGB(255, 255, 255)
        slider.TextScaled = true
        slider.Font = Enum.Font.Gotham
        slider.Parent = frame

        local value = defaultVal
        slider.MouseButton1Click:Connect(function()
            local newVal = value + 10
            if newVal > maxVal then newVal = minVal end
            value = newVal
            slider.Text = tostring(value)
            callback(value)
        end)
        return slider
    end

    -- КНОПКИ
    createButton("👁️ ESP", 0.10, function(s) toggles.esp = s updateESP() end)
    createButton("🔴 Murderer", 0.16, function(s) toggles.showMurderer = s end)
    createButton("🟢 Sheriff", 0.22, function(s) toggles.showSheriff = s end)
    createButton("⚪ Innocent", 0.28, function(s) toggles.showInnocent = s end)
    createButton("🎯 Trigger Bot", 0.34, toggleTriggerBot)
    createButton("🐇 Bunny Hop", 0.40, toggleBunnyHop)
    createButton("🦘 Super Jump", 0.46, toggleJump)
    createButton("🌀 Noclip", 0.52, toggleNoclip)
    createButton("📍 Teleport", 0.58, toggleTeleport)
    createButton("✈️ Fly", 0.64, toggleFly)
    createButton("🔄 Spin Bot", 0.70, toggleSpinBot)
    createButton("💥 Fling", 0.76, toggleFling)
    createButton("🔫 Shot Murderer [H]", 0.82, toggleShotMurderer)
    createButton("🔫 Авто-подбор", 0.88, function(s) settings.autoPickupGun = s print(s and "✅ Авто-подбор включён" or "❌ Авто-подбор выключен") end)
    createButton("🔫 Забрать пистолет [G]", 0.94, function() pickupGunFromMap() end)
    createButton("📊 FPS", 1.00, toggleFps)

    -- СЛАЙДЕРЫ
    createSlider("🐇 Bunny Speed", 1.06, 20, 300, settings.bunnyhop, function(v) settings.bunnyhop = v end)
    createSlider("✈️ Fly Speed", 1.12, 10, 200, settings.flyspeed, function(v) settings.flyspeed = v end)
    createSlider("🦘 Jump Power", 1.18, 50, 300, settings.jumppower, function(v) settings.jumppower = v end)

    print("✅ Ryzen Hub загружен!")
end

-- ===== ОТКРЫТИЕ ПО ПРАВОМУ SHIFT =====
game:GetService("UserInputService").InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.RightShift then
        if gui and gui.Parent then
            gui:Destroy()
            gui = nil
        else
            createRyzenHub()
        end
    end
end)

-- Обработчик клавиши H (убийство мардера)
game:GetService("UserInputService").InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.H then
        killMurderer()
    end
end)

-- Обработчик клавиши G (забрать пистолет)
game:GetService("UserInputService").InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.G then
        pickupGunFromMap()
    end
end)

createRyzenHub()
print("✅ Ryzen Hub готов к работе! Нажмите ПРАВЫЙ SHIFT для открытия меню.")
