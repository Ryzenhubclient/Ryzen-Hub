-- 1. АНИМЕ НЕБО
local function createAnimeSky()

    sky.SkyboxFt = "rbxassetid://7074881348"

-- 2. ТУМАН
local function createFog()
    if not toggles.fog then
        lighting.FogEnd = 1000
        lighting.FogColor = Color3.fromRGB(255, 255, 255)
        return
    end
    lighting.FogEnd = settings.fogEnd
    lighting.FogColor = settings.fogColor
end

-- 3. АТМОСФЕРА (цветное свечение)
local function createAtmosphere()
    if visualEffects.atmosphere then visualEffects.atmosphere:Destroy() end
    if not toggles.atmosphere then return end
    
    local atmo = Instance.new("Atmosphere")
    atmo.Parent = lighting
    atmo.Color = settings.atmosphereColor
    atmo.Density = 0.3
    atmo.Offset = 0.1
    atmo.Decay = Color3.fromRGB(100, 50, 255)
    visualEffects.atmosphere = atmo
end

-- 4. BLOOM (свечение)

-- 5. ЦВЕТОКОРРЕКЦИЯ (аниме-фильтр)

    cc.Parent = lighting

-- 6. ОБЛАКА (частицы)
local function createClouds()
    if visualEffects.clouds then 
        for _, v in pairs(visualEffects.clouds) do v:Destroy() end
        visualEffects.clouds = {}
    end
    if not toggles.animeSky then return end
    
    for i = 1, 15 do
        local cloud = Instance.new("Part")
        cloud.Size = Vector3.new(math.random(20, 50), 3, math.random(20, 50))
        cloud.Position = Vector3.new(math.random(-500, 500), math.random(100, 300), math.random(-500, 500))
        cloud.Anchored = true
        cloud.CanCollide = false
        cloud.Material = Enum.Material.Neon
        cloud.BrickColor = BrickColor.new("Pastel Blue")
        cloud.Transparency = 0.7
        cloud.Parent = workspace
        
        local cloud2 = Instance.new("Part")
        cloud2.Size = Vector3.new(math.random(30, 60), 2, math.random(30, 60))
        cloud2.Position = cloud.Position + Vector3.new(math.random(-20, 20), math.random(-5, 5), math.random(-20, 20))
        cloud2.Anchored = true
        cloud2.CanCollide = false
        cloud2.Material = Enum.Material.Neon
        cloud2.BrickColor = BrickColor.new("Pastel Blue")
        cloud2.Transparency = 0.8
        cloud2.Parent = workspace
        
        table.insert(visualEffects.clouds, cloud)
        table.insert(visualEffects.clouds, cloud2)
    end
end

-- ===== ОПРЕДЕЛЕНИЕ РОЛИ =====
local function getRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    for _, tool in ipairs(plr.Character:GetChildren()) do
        if tool:IsA("Tool") then
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    if plr.Backpack then
        for _, tool in ipairs(plr.Backpack:GetChildren()) do
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    return "Innocent"
end

-- ===== ОСНОВНЫЕ ФУНКЦИИ =====
function updateESP()
    for _, v in pairs(espObjects) do v:Destroy() end
    espObjects = {}
    if not toggles.esp then return end
    
    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local esp = Instance.new("BillboardGui")
            esp.Size = UDim2.new(0, 200, 0, 50)
            esp.AlwaysOnTop = true
            esp.Parent = plr.Character.HumanoidRootPart
            
            local label = Instance.new("TextLabel")
            label.Size = UDim2.new(1, 0, 1, 0)
            label.BackgroundTransparency = 1
            label.TextColor3 = Color3.fromRGB(255, 100, 255)
            label.TextScaled = true
            label.Font = Enum.Font.GothamBold
            
            if settings.espMode == "Имена" then
                label.Text = plr.Name
            elseif settings.espMode == "Роли" then
                label.Text = getRole(plr)
            else
                local dist = (plr.Character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
                label.Text = math.floor(dist) .. "м"
            end
            label.Parent = esp
            
            table.insert(espObjects, esp)
        end
    end
end

function toggleSilentAim(s) toggles.silentAim = s end
function toggleHitbox(s) 
    toggles.hitbox = s
    if not s then
        for _, plr in ipairs(game.Players:GetPlayers()) do
            if plr ~= player and plr.Character then
                for _, part in ipairs(plr.Character:GetChildren()) do
                    if part:IsA("BasePart") then
                        part.Size = Vector3.new(2, 2, 2)
                    end
                end
            end
        end
    end
end
