-- ===== ОПРЕДЕЛЕНИЕ РОЛИ =====
-- ============================================
local function getRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    for _, tool in ipairs(plr.Character:GetChildren()) do
        if tool:IsA("Tool") then
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    if plr.Backpack then
        for _, tool in ipairs(plr.Backpack:GetChildren()) do
            local n = tool.Name:lower()
            if n:find("knife") or n:find("dagger") then return "Murderer" end
            if n:find("gun") or n:find("pistol") or n:find("revolver") then return "Sheriff" end
        end
    end
    return "Innocent"
end

local function getRoleColor(role)
    if role == "Murderer" then return Color3.fromRGB(255, 0, 0) end
    if role == "Sheriff" then return Color3.fromRGB(0, 100, 255) end
    return Color3.fromRGB(0, 255, 0)
end

-- ============================================
-- ===== ESP (ПОЛНАЯ ВЕРСИЯ) =====
-- ============================================
function updateESP()
    for _, v in pairs(espObjects) do pcall(function() v:Destroy() end) end
    espObjects = {}
    for _, v in pairs(tracerObjects) do pcall(function() v:Destroy() end) end
    tracerObjects = {}
    if not toggles.esp then return end

    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local role = getRole(plr)
            local color = getRoleColor(role)
            local root = plr.Character.HumanoidRootPart

            if toggles.boxEsp then
                local box = Instance.new("BoxHandleAdornment")
                box.Size = Vector3.new(3, 5, 1)
                box.Position = root.Position
                box.CFrame = root.CFrame
                box.Color3 = color
                box.Transparency = 0.3
                box.ZIndex = 0
                box.Parent = root
                table.insert(espObjects, box)
            end

            local esp = Instance.new("BillboardGui")
            esp.Size = UDim2.new(0, 250, 0, 50)
            esp.AlwaysOnTop = true
            esp.Parent = root

            local label = Instance.new("TextLabel")
            label.Size = UDim2.new(1, 0, 1, 0)
            label.BackgroundTransparency = 1
            label.TextColor3 = color
            label.TextScaled = true
            label.Font = Enum.Font.GothamBold
            label.Text = settings.espMode == "Имена" and plr.Name or role
            label.Parent = esp
            table.insert(espObjects, esp)

            if toggles.tracers and player.Character then
                local playerRoot = player.Character:FindFirstChild("HumanoidRootPart")
                if playerRoot then
                    local tracer = Instance.new("LineHandleAdornment")
                    tracer.Size = 0.1
                    tracer.Points = {playerRoot.Position, root.Position}
                    tracer.Color3 = color
                    tracer.Transparency = 0.5
                    tracer.ZIndex = 0
                    tracer.Parent = root
                    table.insert(tracerObjects, tracer)
                end
            end
        end
    end

    if settings.espGun then
        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("Tool") and (v.Name:lower():find("gun") or v.Name:lower():find("pistol")) then
                if v.Parent ~= player.Character then
                    local handle = v:FindFirstChild("Handle")
                    if handle then
                        local esp = Instance.new("BillboardGui")
                        esp.Size = UDim2.new(0, 100, 0, 30)
                        esp.AlwaysOnTop = true
                        esp.Parent = handle
                        local label = Instance.new("TextLabel")
                        label.Size = UDim2.new(1, 0, 1, 0)
                        label.BackgroundTransparency = 1
                        label.TextColor3 = Color3.fromRGB(255, 255, 0)
                        label.Text = "🔫 ПИСТОЛЕТ"
                        label.TextScaled = true
                        label.Font = Enum.Font.GothamBold
                        label.Parent = esp
                        table.insert(espObjects, esp)
                    end
                end
            end
        end
    end
end

-- ============================================
-- ===== FLY =====
-- ============================================
function toggleFly(state)
    toggles.fly = state
    if flyConnection then flyConnection:Disconnect() flyConnection = nil end
    if not state then
        local char = player.Character
        if char then
            local humanoid = char:FindFirstChild("Humanoid")
            if humanoid then humanoid.PlatformStand = false end
        end
        return
    end
    local char = player.Character
    if not char then return end
    local humanoid = char:FindFirstChild("Humanoid")
    if humanoid then humanoid.PlatformStand = true end
    flyConnection = runService.Heartbeat:Connect(function()
        if not player.Character then return end
        local root = player.Character:FindFirstChild("HumanoidRootPart")
        if not root then return end
        local direction = Vector3.new(0, 0, 0)
        if uis:IsKeyDown(Enum.KeyCode.W) then direction = direction + root.CFrame.LookVector end
        if uis:IsKeyDown(Enum.KeyCode.S) then direction = direction - root.CFrame.LookVector end
        if uis:IsKeyDown(Enum.KeyCode.A) then direction = direction - root.CFrame.RightVector end
        if uis:IsKeyDown(Enum.KeyCode.D) then direction = direction + root.CFrame.RightVector end
        if uis:IsKeyDown(Enum.KeyCode.Space) then direction = direction + Vector3.new(0, 1, 0) end
        if uis:IsKeyDown(Enum.KeyCode.LeftShift) then direction = direction - Vector3.new(0, 1, 0) end
        if direction.Magnitude > 0 then
            direction = direction.Unit * settings.flySpeed
        end
        root.Velocity = direction
    end)
end

-- ============================================
-- ===== KILL AURA =====
-- ============================================
function toggleKillAura(state)
    toggles.killAura = state
    if killAuraConnection then killAuraConnection:Disconnect() killAuraConnection = nil end
    if not state then return end
    killAuraConnection = runService.Heartbeat:Connect(function()
        if not player.Character then return end
        local root = player.Character:FindFirstChild("HumanoidRootPart")
        if not root then return end
        for _, plr in ipairs(game.Players:GetPlayers()) do
            if plr ~= player and plr.Character then
                local targetRoot = plr.Character:FindFirstChild("HumanoidRootPart")
                if targetRoot and (targetRoot.Position - root.Position).Magnitude <= settings.killAuraRange then
                    local humanoid = plr.Character:FindFirstChild("Humanoid")
                    if humanoid then humanoid.Health = 0 end
                end
            end
        end
    end)
end

function killAll()
    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character then
            local humanoid = plr.Character:FindFirstChild("Humanoid")
            if humanoid then
                if toggles.killOnlySheriff then
                    if getRole(plr) == "Sheriff" then humanoid.Health = 0 end

                end
            end
        end
    end
end

-- ============================================
-- ===== KNIFE AIMBOT =====
-- ============================================
function toggleKnifeAimbot(state)
    toggles.knifeAimbot = state
    if aimbotConnection then aimbotConnection:Disconnect() aimbotConnection = nil end
    if not state then return end
    aimbotConnection = runService.Heartbeat:Connect(function()
        if not player.Character then return end
        local root = player.Character:FindFirstChild("HumanoidRootPart")
        if not root then return end
        local knife = player.Character:FindFirstChildOfClass("Tool")
        if not knife or not knife.Name:lower():find("knife") then return end
        local target = nil
        local targetDist = 50
        for _, plr in ipairs(game.Players:GetPlayers()) do
            if plr ~= player and plr.Character then
                local targetRoot = plr.Character:FindFirstChild("HumanoidRootPart")
                if targetRoot then
                    local dist = (targetRoot.Position - root.Position).Magnitude
                    if dist < targetDist then
                        targetDist = dist
                        target = plr
                    end
                end
            end
        end
        if target and target.Character then
            local targetRoot = target.Character:FindFirstChild("HumanoidRootPart")
            if targetRoot then
                local direction = (targetRoot.Position - root.Position).Unit
                root.CFrame = CFrame.lookAt(root.Position, root.Position + direction)
                mouse1click()
            end
        end
    end)
end

function teleportToRole(role)
    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and getRole(plr) == role and plr.Character then
            local root = plr.Character:FindFirstChild("HumanoidRootPart")
            if root and player.Character then

                    charRoot.CFrame = root.CFrame * CFrame.new(0, 2, 0)
                end
            end
        end
    end

function openPlayerMenu(plr)
    if playerMenu then playerMenu:Destroy() end
    playerMenu = Instance.new("ScreenGui")
    playerMenu.Name = "PlayerMenu"
    playerMenu.Parent = game.CoreGui

    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(0, 250, 0, 300)
    frame.Position = UDim2.new(0.5, -125, 0.5, -150)
    frame.BackgroundColor3 = Color3.fromRGB(20, 0, 30)
    frame.BackgroundTransparency = 0.1
    frame.BorderSizePixel = 2
    frame.BorderColor3 = Color3.fromRGB(200, 100, 255)
    frame.Active = true
    frame.Draggable = true
    frame.Parent = playerMenu

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1,0,0,30)
    title.BackgroundColor3 = Color3.fromRGB(40, 0, 50)
    title.BorderSizePixel = 0
    title.Text = plr.Name
    title.TextColor3 = Color3.fromRGB(200, 100, 255)
    title.TextScaled = true
    title.Font = Enum.Font.GothamBold
    title.Parent = frame

    local close = Instance.new("TextButton")
    close.Size = UDim2.new(0,25,0,25)
    close.Position = UDim2.new(1,-30,0,2)
    close.BackgroundColor3 = Color3.fromRGB(80,0,80)
    close.BorderSizePixel = 0
    close.Text = "✕"
    close.TextColor3 = Color3.fromRGB(255,255,255)
    close.TextScaled = true
    close.Font = Enum.Font.GothamBold
    close.Parent = frame
    close.MouseButton1Click:Connect(function() playerMenu:Destroy() end)

    local buttons = {"Teleport", "Fling", "Kill", "Spectate"}
    local y = 0.15
    for _, name in ipairs(buttons) do
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(0.8,0,0,30)
        btn.Position = UDim2.new(0.1,0,y,0)
        btn.BackgroundColor3 = Color3.fromRGB(40,0,50)
        btn.BorderSizePixel = 1
        btn.BorderColor3 = Color3.fromRGB(200,100,255)
        btn.Text = name
        btn.TextColor3 = Color3.fromRGB(255,255,255)
        btn.TextScaled = true
        btn.Font = Enum.Font.Gotham
        btn.Parent = frame
        
        btn.MouseButton1Click:Connect(function()
            if name == "Teleport" then
                if plr.Character then
                    local root = plr.Character:FindFirstChild("HumanoidRootPart")
                    if root and player.Character then
                        local charRoot = player.Character:FindFirstChild("HumanoidRootPart")
                        if charRoot then charRoot.CFrame = root.CFrame * CFrame.new(0, 2, 0) end
                    end
                end
            elseif name == "Fling" then
                if plr.Character then
                    local root = plr.Character:FindFirstChild("HumanoidRootPart")
                    if root and player.Character then
                        local charRoot = player.Character:FindFirstChild("HumanoidRootPart")
                        if charRoot then
                            local direction = (root.Position - charRoot.Position).Unit
                            root.Velocity = direction * 1000 + Vector3.new(0, 500, 0)
                        end
                    end
                end
            elseif name == "Kill" then
                if plr.Character then
                    local humanoid = plr.Character:FindFirstChild("Humanoid")
                    if humanoid then humanoid.Health = 0 end
                end
            elseif name == "Spectate" then
                workspace.CurrentCamera.CameraSubject = plr.Character
            end
            playerMenu:Destroy()
        end)
        y = y + 0.12
    end
end

-- ============================================
-- ===== SHADERS =====
-- ============================================
function setShader(shaderType)
    for _, v in pairs(lighting:GetChildren()) do
        if v:IsA("BloomEffect") or v:IsA("ColorCorrectionEffect") or v:IsA("SunRaysEffect") then
            v:Destroy()
        end
    end
    if shaderType == "RTX" then
        local bloom = Instance.new("BloomEffect")
        bloom.Intensity = 0.3
        bloom.Size = 50
        bloom.Threshold = 0.5
        bloom.Parent = lighting
        local cc = Instance.new("ColorCorrectionEffect")
        cc.Brightness = 0.1
        cc.Contrast = 0.2
        cc.Saturation = 0.3
        cc.Parent = lighting
    elseif shaderType == "Cinematic" then
        local cc = Instance.new("ColorCorrectionEffect")
        cc.Brightness = -0.1
        cc.Contrast = 0.3
        cc.Saturation = 0.1
        cc.TintColor = Color3.fromRGB(255, 200, 150)
        cc.Parent = lighting
    elseif shaderType == "Sunset Vibes" then
        local bloom = Instance.new("BloomEffect")
        bloom.Intensity = 0.5
        bloom.Size = 100
        bloom.Threshold = 0.3
        bloom.Parent = lighting
        local cc = Instance.new("ColorCorrectionEffect")
        cc.TintColor = Color3.fromRGB(255, 150, 100)
        cc.Parent = lighting
    elseif shaderType == "Midnight" then
        local cc = Instance.new("ColorCorrectionEffect")
        cc.Brightness = -0.2
        cc.Contrast = 0.1
        cc.Saturation = -0.1
        cc.TintColor = Color3.fromRGB(50, 50, 100)
        cc.Parent = lighting
    end
end

local function createTrail(origin, target)
    local trail = Instance.new("Trail")

    trail.Lifetime = 0.5
    trail.MinLength = 0.1
    trail.Transparency = NumberSequence.new(0, 1)

end

local function createMiniMenu(funcName, x, y)
    if miniMenu then miniMenu:Destroy() end
    miniMenu = Instance.new("ScreenGui")
    miniMenu.Name = "MiniMenu"
    miniMenu.Parent = game.CoreGui

    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(0, 300, 0, 400)
    frame.Position = UDim2.new(0, x, 0, y)
    frame.BackgroundColor3 = Color3.fromRGB(20, 0, 30)
    frame.BackgroundTransparency = 0.1
    frame.BorderSizePixel = 2
    frame.BorderColor3 = Color3.fromRGB(200, 100, 255)
    frame.Active = true
    frame.Draggable = true
    frame.Parent = miniMenu

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1,0,0,30)
    title.BackgroundColor3 = Color3.fromRGB(40, 0, 50)
    title.BorderSizePixel = 0
    title.Text = "⚙️ " .. funcName
    title.TextColor3 = Color3.fromRGB(200, 100, 255)
    title.TextScaled = true
    title.Font = Enum.Font.GothamBold
    title.Parent = frame

    local close = Instance.new("TextButton")
    close.Size = UDim2.new(0,25,0,25)
    close.Position = UDim2.new(1,-30,0,2)
    close.BackgroundColor3 = Color3.fromRGB(80,0,80)
    close.BorderSizePixel = 0
    close.Text = "✕"
    close.TextColor3 = Color3.fromRGB(255,255,255)
    close.TextScaled = true
    close.Font = Enum.Font.GothamBold
    close.Parent = frame
    close.MouseButton1Click:Connect(function() miniMenu:Destroy() end)

    local function createSlider(txt, y, min, max, def, cb)
        local fr = Instance.new("Frame")
        fr.Size = UDim2.new(0.9,0,0,25)
        fr.Position = UDim2.new(0.05,0,y,0)
        fr.BackgroundColor3 = Color3.fromRGB(30,0,40)
        fr.BorderSizePixel = 0
        fr.Parent = frame

        local lbl = Instance.new("TextLabel")
        lbl.Size = UDim2.new(0.4,0,1,0)
        lbl.Position = UDim2.new(0.02,0,0,0)
        lbl.BackgroundTransparency = 1
        lbl.Text = txt
        lbl.TextColor3 = Color3.fromRGB(200,180,255)
        lbl.TextScaled = true
        lbl.Font = Enum.Font.Gotham
        lbl.Parent = fr

        local sl = Instance.new("TextButton")
        sl.Size = UDim2.new(0.4,0,0.7,0)
        sl.Position = UDim2.new(0.45,0,0.15,0)
        sl.BackgroundColor3 = Color3.fromRGB(50,0,60)
        sl.BorderSizePixel = 1
        sl.BorderColor3 = Color3.fromRGB(200,100,255)
        sl.Text = tostring(def)
        sl.TextColor3 = Color3.fromRGB(255,255,255)
        sl.TextScaled = true
        sl.Font = Enum.Font.Gotham
        sl.Parent = fr
        local val = def
        sl.MouseButton1Click:Connect(function()
            val = val + 1
            if val > max then val = min end
            sl.Text = tostring(val)
            cb(val)
        end)
    end
