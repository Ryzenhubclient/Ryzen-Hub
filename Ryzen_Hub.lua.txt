-- RYZEN HUB - ИСПРАВЛЕННАЯ ВЕРСИЯ (КВАДРАТНОЕ МЕНЮ)
local player = game.Players.LocalPlayer
local mouse = player:GetMouse()

-- ===== НАСТРОЙКИ =====
local settings = {
    speedglitch = 100,
    flyspeed = 50,
    jumppower = 120,
    spinSpeed = 10,
    aimbotFov = 150,
}

local toggles = {
    esp = false,
    triggerbot = false,
    speedglitch = false,
    jump = false,
    noclip = false,
    teleport = false,
    fly = false,
    showFps = false,
    spinBot = false,
    fling = false,
    shotMurderer = false,
    aimbot = false,
    showMurderer = true,
    showSheriff = true,
    showInnocent = false,
}

local gui = nil
local espObjects = {}
local fpsLabel = nil
local fovCircle = nil
local connections = {}

-- ===== ФУНКЦИЯ ОПРЕДЕЛЕНИЯ РОЛИ =====
local function getRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    for _, tool in ipairs(plr.Character:GetChildren()) do
        if tool:IsA("Tool") then
            local name = tool.Name:lower()
            if name:find("knife") or name:find("dagger") then return "Murderer" end
            if name:find("gun") or name:find("pistol") or name:find("revolver") then return "Sheriff" end
        end
    end
    if plr.Backpack then
        for _, tool in ipairs(plr.Backpack:GetChildren()) do
            if tool:IsA("Tool") then
                local name = tool.Name:lower()
                if name:find("knife") or name:find("dagger") then return "Murderer" end
                if name:find("gun") or name:find("pistol") or name:find("revolver") then return "Sheriff" end
            end
        end
    end
    return "Innocent"
end

-- ===== ESP =====
local function updateESP()
    for _, obj in ipairs(espObjects) do
        pcall(function() obj:Destroy() end)
    end
    espObjects = {}
    if not toggles.esp then return end

    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr == player then continue end
        if not plr.Character or not plr.Character:FindFirstChild("Head") then continue end
        local role = getRole(plr)
        local show = false
        if role == "Murderer" and toggles.showMurderer then show = true end
        if role == "Sheriff" and toggles.showSheriff then show = true end
        if role == "Innocent" and toggles.showInnocent then show = true end
        if not show then continue end
        local color = role == "Murderer" and Color3.fromRGB(255,0,0) or role == "Sheriff" and Color3.fromRGB(0,100,255) or Color3.fromRGB(0,255,0)
        local esp = Instance.new("BillboardGui")
        esp.Name = "ESP_Ryzen"
        esp.Size = UDim2.new(0, 150, 0, 40)
        esp.AlwaysOnTop = true
        esp.Parent = plr.Character.Head
        table.insert(espObjects, esp)
        local text = Instance.new("TextLabel")
        text.Size = UDim2.new(1, 0, 1, 0)
        text.BackgroundTransparency = 1
        text.Text = plr.Name .. " [" .. role .. "]"
        text.TextColor3 = color
        text.TextScaled = true
        text.Font = Enum.Font.GothamBold
        text.Parent = esp
        table.insert(espObjects, text)
        local highlight = Instance.new("Highlight")
        highlight.Adornee = plr.Character
        highlight.FillColor = color
        highlight.FillTransparency = 0.5
        highlight.OutlineColor = color
        highlight.Parent = plr.Character
        table.insert(espObjects, highlight)
    end
end

-- ===== AIMBOT =====
local function toggleAimbot(state)
    toggles.aimbot = state
    if state then
        if connections.aimbot then connections.aimbot:Disconnect() end
        connections.aimbot = game:GetService("RunService").Heartbeat:Connect(function()
            if not toggles.aimbot then return end
            if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
            local target, targetDist = nil, math.huge
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character and plr.Character:FindFirstChild("Head") then
                    local dist = (plr.Character.Head.Position - player.Character.HumanoidRootPart.Position).Magnitude
                    if dist < targetDist and dist < settings.aimbotFov then
                        target = plr
                        targetDist = dist
                    end
                end
            end
            if target then
                player.Character.HumanoidRootPart.CFrame = CFrame.new(player.Character.HumanoidRootPart.Position, target.Character.Head.Position)
            end
        end)
        -- FOV круг
        if fovCircle then fovCircle:Destroy() end
        fovCircle = Instance.new("Frame")
        fovCircle.Size = UDim2.new(0, settings.aimbotFov * 2, 0, settings.aimbotFov * 2)
        fovCircle.Position = UDim2.new(0.5, -settings.aimbotFov, 0.5, -settings.aimbotFov)
        fovCircle.BackgroundColor3 = Color3.fromRGB(255,255,255)
        fovCircle.BackgroundTransparency = 0.85
        fovCircle.BorderSizePixel = 1
        fovCircle.BorderColor3 = Color3.fromRGB(255,0,0)
        fovCircle.Active = false
        fovCircle.Parent = game.CoreGui
        local corner = Instance.new("UICorner")
        corner.CornerRadius = UDim.new(1,0)
        corner.Parent = fovCircle
    else
        if connections.aimbot then connections.aimbot:Disconnect() end
        if fovCircle then fovCircle:Destroy() end
    end
end

-- ===== TRIGGER BOT =====
local function toggleTriggerBot(state)
    toggles.triggerbot = state
    if state then
        if connections.triggerbot then connections.triggerbot:Disconnect() end
        connections.triggerbot = game:GetService("RunService").Heartbeat:Connect(function()
            if not toggles.triggerbot then return end
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= player and plr.Character and plr.Character:FindFirstChild("Humanoid") then
                    if getRole(plr) == "Murderer" then
                        pcall(function() plr.Character.Humanoid.Health = 0 end)
                    end
                end
            end
        end)
    else
        if connections.triggerbot then connections.triggerbot:Disconnect() end
    end
end

-- ===== SHOT MURDERER =====
local function toggleShotMurderer(state)
    toggles.shotMurderer = state
end
local function killMurderer()
    if not toggles.shotMurderer then return end
    for _, plr in ipairs(game.Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("Humanoid") then
            if getRole(plr) == "Murderer" then
                pcall(function() plr.Character.Humanoid.Health = 0 end)
                return
            end
        end
    end
end

-- ===== SPEED GLITCH =====
local function toggleSpeedGlitch(state)
    toggles.speedglitch = state
    if state then
        if connections.speedglitch then connections.speedglitch:Disconnect() end
        connections.speedglitch = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.speedglitch then return end
            if player.Character and player.Character:FindFirstChild("Humanoid") then
                local humanoid = player.Character.Humanoid
                if humanoid:GetState() == Enum.HumanoidStateType.Jumping or humanoid:GetState() == Enum.HumanoidStateType.Falling then
                    humanoid.WalkSpeed = settings.speedglitch
                else
                    humanoid.WalkSpeed = 16
                end
            end
        end)
    else
        if connections.speedglitch then connections.speedglitch:Disconnect() end
        if player.Character and player.Character:FindFirstChild("Humanoid") then
            player.Character.Humanoid.WalkSpeed = 16
        end
    end
end

-- ===== SUPER JUMP =====
local function toggleJump(state)
    toggles.jump = state
    if player.Character and player.Character:FindFirstChild("Humanoid") then
        player.Character.Humanoid.JumpPower = state and settings.jumppower or 50
    end
end

-- ===== NOCLIP =====
local function toggleNoclip(state)
    toggles.noclip = state
    if state then
        if connections.noclip then connections.noclip:Disconnect() end
        connections.noclip = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.noclip then return end
            if player.Character then
                for _, part in ipairs(player.Character:GetChildren()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end)
    else
        if connections.noclip then connections.noclip:Disconnect() end
        if player.Character then
            for _, part in ipairs(player.Character:GetChildren()) do
                if part:IsA("BasePart") then part.CanCollide = true end
            end
        end
    end
end

-- ===== TELEPORT =====
local function toggleTeleport(state)
    toggles.teleport = state
end
mouse.Button1Down:Connect(function()
    if toggles.teleport and mouse.Target then
        local pos = mouse.Target.Position
        if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            player.Character.HumanoidRootPart.CFrame = CFrame.new(pos)
        end
    end
end)

-- ===== FLY =====
local function toggleFly(state)
    toggles.fly = state
    if state then
        if connections.fly then connections.fly:Disconnect() end
        connections.fly = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.fly then return end
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                local root = player.Character.HumanoidRootPart
                local dir = Vector3.new(0,0,0)
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.W) then dir = dir + root.CFrame.LookVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.S) then dir = dir - root.CFrame.LookVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.A) then dir = dir - root.CFrame.RightVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.D) then dir = dir + root.CFrame.RightVector end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
                if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.LeftShift) then dir = dir - Vector3.new(0,1,0) end
                root.Velocity = dir * settings.flyspeed
            end
        end)
    else
        if connections.fly then connections.fly:Disconnect() end
        if player.Character and player.Character:FindFirstChild("Humanoid") then
            player.Character.Humanoid.PlatformStand = false
        end
    end
end

-- ===== SPIN BOT =====
local function toggleSpinBot(state)
    toggles.spinBot = state
    if state then
        if connections.spin then connections.spin:Disconnect() end
        connections.spin = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.spinBot then return end
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                player.Character.HumanoidRootPart.CFrame = player.Character.HumanoidRootPart.CFrame * CFrame.Angles(0, math.rad(settings.spinSpeed), 0)
            end
        end)

        if connections.spin then connections.spin:Disconnect() end
    end
end

-- ===== FLING =====
local function toggleFling(state)
    toggles.fling = state
    if state then
        if connections.fling then connections.fling:Disconnect() end
        connections.fling = game:GetService("RunService").Stepped:Connect(function()
            if not toggles.fling then return end
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                for _, plr in ipairs(game.Players:GetPlayers()) do
                    if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                        local dist = (plr.Character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
                        if dist < 10 then
                            pcall(function() plr.Character.HumanoidRootPart.Velocity = Vector3.new(0, 1000, 0) end)
                        end
                    end
                end
            end
        end)
    else
        if connections.fling then connections.fling:Disconnect() end
    end
end

-- ===== FPS =====
local function toggleFps(state)
    toggles.showFps = state
    if state then
        if fpsLabel then fpsLabel:Destroy() end
        local sg = Instance.new("ScreenGui")
        sg.Name = "FPSDisplay"
        sg.Parent = game.CoreGui
        fpsLabel = Instance.new("TextLabel")
        fpsLabel.Size = UDim2.new(0, 100, 0, 30)
        fpsLabel.Position = UDim2.new(0, 10, 0, 10)
        fpsLabel.BackgroundColor3 = Color3.fromRGB(0,0,0)
        fpsLabel.BackgroundTransparency = 0.3
        fpsLabel.BorderSizePixel = 0
        fpsLabel.Text = "FPS: 0"
        fpsLabel.TextColor3 = Color3.fromRGB(255,255,255)
        fpsLabel.TextScaled = true
        fpsLabel.Font = Enum.Font.GothamBold
        fpsLabel.Parent = sg
        local frames = 0
        local last = tick()
        game:GetService("RunService").Heartbeat:Connect(function()
            frames = frames + 1
            if tick() - last >= 1 then
                if fpsLabel then fpsLabel.Text = "FPS: " .. frames end
                frames = 0
                last = tick()
            end
        end)
    else
        if fpsLabel then fpsLabel:Destroy() end
    end
end

-- ===== ПОДБОР ПИСТОЛЕТА =====
local function pickupGun()
    if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
    local root = player.Character.HumanoidRootPart
    for _, item in ipairs(workspace:GetDescendants()) do
        if item:IsA("Tool") then
            local name = item.Name:lower()
            if name:find("gun") or name:find("pistol") or name:find("revolver") then
                if item.Parent ~= player.Character and item.Parent ~= player.Backpack then
                    local dist = (item.Position - root.Position).Magnitude
                    if dist < 100 then
                        pcall(function()
                            root.CFrame = CFrame.new(item.Position + Vector3.new(0,2,0))
                            task.wait(0.2)
                            item.Parent = player.Character
                        end)
                        return
                    end
                end
            end
        end
    end
end

    close.TextScaled = true

        btn.Position = UDim2.new(0.075, 0, y, 0)

-- ===== ОТКРЫТИЕ ПО ПРАВОМУ SHIFT =====
game:GetService("UserInputService").InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.RightShift then
        if gui and gui.Parent then
            gui:Destroy()
            gui = nil
        else
            createGUI()
        end
    end
end)

game:GetService("UserInputService").InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.H then
        killMurderer()
    end
    if input.KeyCode == Enum.KeyCode.G then
        pickupGun()
    end
end)
