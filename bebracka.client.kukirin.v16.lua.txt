local function getRide()
	local char = LocalPlayer.Character
	if not char then return nil, nil end
	local hum = char:FindFirstChildOfClass("Humanoid")
	local seat = hum and hum.SeatPart
	if seat then return seat, seat.AssemblyRootPart or seat end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if hrp then
		for _, obj in ipairs(hrp:GetChildren()) do
			if obj:IsA("Weld") or obj:IsA("WeldConstraint") or obj:IsA("Motor6D") then
				local other = nil
				if obj.Part0 == hrp then other = obj.Part1
				elseif obj.Part1 == hrp then other = obj.Part0 end
				if other and not other:IsDescendantOf(char) then
					return nil, other.AssemblyRootPart or other
				end
			end
		end
	end
	local tool = char:FindFirstChildOfClass("Tool")
	local handle = tool and tool:FindFirstChild("Handle")
	if handle then return nil, handle.AssemblyRootPart or handle end
	if hrp then
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = { char }
		local hit = workspace:Raycast(hrp.Position, Vector3.new(0, -12, 0), params)
		if hit and hit.Instance and hit.Instance:IsA("BasePart") then
			local hitPart = hit.Instance
			local model = hitPart:FindFirstAncestorOfClass("Model")
			local root = hitPart.AssemblyRootPart or hitPart
			local isVehicle = (not hitPart.Anchored) or (not root.Anchored)
			if isVehicle and model and model ~= workspace and model ~= char then
				local seatPart = model:FindFirstChildOfClass("VehicleSeat") or model:FindFirstChildOfClass("Seat")
				if seatPart then return seatPart, seatPart.AssemblyRootPart or seatPart end
				return nil, root
			end
		end
	end
	return nil, hrp
end

local function hasSpeedInput(seat)
	local okF, f = pcall(function() return UserInputService:GetFocusedTextBox() end)
	if okF and f then return false end
	if scriptWDown then return true end
	local okW, w = pcall(function() return UserInputService:IsKeyDown(Enum.KeyCode.W) end)
	if okW and w then return true end
	local okU, u = pcall(function() return UserInputService:IsKeyDown(Enum.KeyCode.Up) end)
	if okU and u then return true end
	if seat and seat:IsA("VehicleSeat") then
		local okT, th = pcall(function() return seat.Throttle end)
		if okT and type(th) == "number" and th > 0.05 then return true end
	end
	return false
end

local GEAR_FACTOR = { eco = 1/3, sport = 2/3, race = 1 }
local NUM_GEAR = { [1] = "eco", [2] = "sport", [3] = "race" }
local detectedGear, detectedSource, gearScanClock = nil, nil, 0

local function gearSpeed(g)
	return math.floor(CONFIG.speed * (GEAR_FACTOR[g] or 1) + 0.5)
end

local function matchGear(v)
	if type(v) ~= "string" then return nil end
	local low = v:lower()
	if low:find("%f[%a]eco%f[%A]") then return "eco" end
	if low:find("%f[%a]sports?%f[%A]") then return "sport" end
	if low:find("%f[%a]rac[ei]n?g?%f[%A]") then return "race" end
	return nil
end

local function gearFromValue(name, v)
	local lname = tostring(name):lower()
	local nameOk = lname:find("mode") or lname:find("gear") or lname:find("speed") or lname:find("drive") or lname:find("profile")
	if type(v) == "string" then
		local trimmed = v:lower():gsub("%s+", "")
		if trimmed == "eco" or trimmed == "sport" or trimmed == "sports" or trimmed == "race" or trimmed == "racing" then
			return matchGear(trimmed)
		end
		if nameOk then return matchGear(v) end
	elseif type(v) == "number" then
		if (lname:find("gear") or lname:find("mode")) and NUM_GEAR[v] then return NUM_GEAR[v] end
	end
	return nil
end

local function scanContainer(inst, deep)
	if not inst then return nil end
	local ok, attrs = pcall(function() return inst:GetAttributes() end)
	if ok and attrs then
		for name, val in pairs(attrs) do
			local g = gearFromValue(name, val)
			if g then return g, inst.Name .. "@" .. name end
		end
	end
	local list = deep and inst:GetDescendants() or inst:GetChildren()
	for i = 1, math.min(#list, 800) do
		local d = list[i]
		if d:IsA("StringValue") or d:IsA("IntValue") or d:IsA("NumberValue") then
			local g = gearFromValue(d.Name, d.Value)
			if g then return g, d:GetFullName() end
		end
	end
	return nil
end

local function detectGear()
	local seat, part = getRide()
	local ref = seat or part
	local model = ref and ref:FindFirstAncestorOfClass("Model")
	local places = {
		{ seat, false }, { part, false }, { model, true },
		{ LocalPlayer, false }, { LocalPlayer.Character, false },
	}
	for _, e in ipairs(places) do
		local g, src = scanContainer(e[1], e[2])
		if g then return g, src end
	end
	return nil
end

local function currentGear()
	if CONFIG.gearMode ~= "auto" then return CONFIG.gearMode, "manual" end
	local now = os.clock()
	if now - gearScanClock > 0.5 then
		gearScanClock = now
		local g, src = nil, nil
		pcall(function() g, src = detectGear() end)
		if g ~= detectedGear then
			print("[bebracka client] gear: " .. tostring(g) .. " (" .. tostring(src) .. ")")
		end
		detectedGear, detectedSource = g, src
	end
	if detectedGear then return detectedGear, "auto" end
	return "race", "default"
end

local function groundNormal(seat, part)
	local char = LocalPlayer.Character
	local ref = seat or part
	local model = ref:FindFirstAncestorOfClass("Model")
	local ignore = { char }
	if model and model ~= char then table.insert(ignore, model)
	else table.insert(ignore, part) end
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = ignore
	local hit = workspace:Raycast(part.Position + Vector3.new(0, 1, 0), Vector3.new(0, -8, 0), params)
	if hit and hit.Normal.Y > 0.5 then return hit.Normal end
	return nil
end

local guardChar, guardOn = nil, false

local function guardHumanoid(char, on)
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	pcall(function()
		hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, not on)
		hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, not on)
	end)
end

local function syncGuard()
	local char = LocalPlayer.Character
	if char ~= guardChar then guardChar = char; guardOn = false end
	if CONFIG.antiFall ~= guardOn then
		guardOn = CONFIG.antiFall
		guardHumanoid(char, guardOn)
	end
end

local function antiFallStep()
	if not CONFIG.antiFall or flyOn or spinning or autoWheelieOn then return end
	local seat, part = getRide()
	if not part or not part.Parent then return end
	local char = LocalPlayer.Character
	if char and part:IsDescendantOf(char) then return end
	local n = groundNormal(seat, part) or Vector3.new(0, 1, 0)
	local ref = seat or part
	local u = ref.CFrame.UpVector
	local ang = math.acos(math.clamp(u:Dot(n), -1, 1))
	if ang > math.rad(1.5) then
		local axis = u:Cross(n)
		if axis.Magnitude > 1e-4 then
			local step = math.min(ang, math.max(ang * 0.6, math.rad(2)))
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			local pivot = hrp and hrp.Position or part.Position
			part.CFrame = CFrame.new(pivot) * CFrame.fromAxisAngle(axis.Unit, step) * CFrame.new(-pivot) * part.CFrame
		end
	end
	local av = part.AssemblyAngularVelocity
	part.AssemblyAngularVelocity = n * av:Dot(n)
end

local function applySpeed(dt)
	if flyOn then speedRampActive = false; speedRampPart = nil; return end
	local seat, part = getRide()
	if not part or not part.Parent then speedRampActive = false; speedRampPart = nil; return end
	if not hasSpeedInput(seat) then speedRampActive = false; speedRampPart = nil; return end
	local look = (seat or part).CFrame.LookVector
	local dir = Vector3.new(look.X, 0, look.Z)
	if dir.Magnitude < 0.01 then speedRampActive = false; speedRampPart = nil; return end
	dir = dir.Unit
	local v = part.AssemblyLinearVelocity
	local normal = nil
	if CONFIG.antiFall then
		normal = groundNormal(seat, part)
		if not normal and math.abs(v.Y) > 8 then return end
	end
	local dir3 = dir
	if normal then
		local proj = dir - normal * dir:Dot(normal)
		if proj.Magnitude > 0.01 then dir3 = proj.Unit end
	end
	local forward = v:Dot(dir3)
	local gear = currentGear()
	local target = gearSpeed(gear)
	if not speedRampActive or speedRampPart ~= part then
		speedRampCurrent = forward; speedRampPart = part; speedRampActive = true
		speedRampStart = forward; speedRampT = 0; speedRampTarget = target
	end
	if speedRampTarget ~= target then
		speedRampStart = speedRampCurrent; speedRampT = 0; speedRampTarget = target
	end
	dt = math.min(dt or (1/60), 0.1)
	local ramp = math.max(0.5, CONFIG.speedRampTime or 7)
	if speedRampCurrent < target then
		speedRampT = math.min(1, speedRampT + dt / ramp)
		local k = speedRampT * speedRampT * (3 - 2 * speedRampT)
		local eased = speedRampStart + (target - speedRampStart) * k
		speedRampCurrent = math.min(target, math.max(speedRampCurrent, eased))
	else
		speedRampCurrent = math.max(target, speedRampCurrent - (CONFIG.speedBrake or 120) * dt)
	end
	local newSpeed = speedRampCurrent
	local yVel = dir3.Y * newSpeed
	if math.abs(dir3.Y) < 0.02 then yVel = v.Y end
	part.AssemblyLinearVelocity = Vector3.new(dir3.X * newSpeed, yVel, dir3.Z * newSpeed)
end

connect(RunService.Heartbeat, function(dt)
	if speedOn then pcall(applySpeed, dt)
	else speedRampActive = false; speedRampPart = nil end
	pcall(syncGuard)
	pcall(antiFallStep)
end)

local function doJump()
	local seat, part = getRide()
	if not part or not part.Parent then return end
	local v = part.AssemblyLinearVelocity
	part.AssemblyLinearVelocity = Vector3.new(v.X, CONFIG.jumpPower, v.Z)
	local bv = Instance.new("BodyVelocity")
	bv.MaxForce = Vector3.new(0, 1e6, 0)
	bv.Velocity = Vector3.new(v.X, CONFIG.jumpPower, v.Z)
	bv.Parent = part
	task.delay(CONFIG.jumpHold, function() if bv then bv:Destroy() end end)
	local char = LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum then pcall(function() hum.Jump = true end) end
end

local spinAutoRotateWas, spinStartClock, spinStartY = nil, 0, 0
local spinBaseCF, spinPivot, spinAngle = nil, nil, 0

local function spinStop()
	spinning = false; spinBaseCF = nil; spinPivot = nil
	local seat, part = getRide()
	if part and part.Parent then pcall(function() part.AssemblyAngularVelocity = Vector3.new(0,0,0) end) end
	local char = LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum and spinAutoRotateWas ~= nil then pcall(function() hum.AutoRotate = spinAutoRotateWas end) end
	spinAutoRotateWas = nil
	if updateUI then pcall(updateUI) end
end

local function spinStart()
	local seat, part = getRide()
	if not part or not part.Parent then return false end
	local char = LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum then
		spinAutoRotateWas = hum.AutoRotate
		pcall(function() hum.AutoRotate = false end)
	end
	local cf0 = part.CFrame
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local pivotPos = hrp and hrp.Position or cf0.Position
	spinning = true; spinStartClock = os.clock(); spinStartY = part.Position.Y
	spinBaseCF = cf0; spinPivot = Vector3.new(pivotPos.X, 0, pivotPos.Z); spinAngle = 0
	return true
end

connect(RunService.Heartbeat, function(dt)
	if not spinning or not spinBaseCF then return end
	local seat, part = getRide()
	if not part or not part.Parent then return end
	local elapsed = os.clock() - spinStartClock
	if elapsed > 0.25 then
		local vy = 0
		pcall(function() vy = part.AssemblyLinearVelocity.Y end)
		if (vy <= 0 and part.Position.Y <= spinStartY + 0.4) or elapsed > 8 then
			spinStop(); return
		end
	end
	spinAngle = spinAngle + CONFIG.spinSpeed * math.min(dt, 0.05)
	pcall(function()
		local v = part.AssemblyLinearVelocity
		part.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
		part.AssemblyLinearVelocity = Vector3.new(0, v.Y, 0)
		local rotated = CFrame.new(spinPivot) * CFrame.Angles(0, spinAngle, 0) * CFrame.new(-spinPivot) * spinBaseCF
		local dy = part.Position.Y - spinBaseCF.Position.Y
		part.CFrame = rotated + Vector3.new(0, dy, 0)
	end)
end)

local function flyInputVector()
	local cam = workspace.CurrentCamera
	if not cam then return Vector3.new(0,0,0) end
	local typing = false
	pcall(function() typing = UserInputService:GetFocusedTextBox() ~= nil end)
	local move = Vector3.new(0,0,0)
	local horizontalKey = false
	if not typing then
		local look = cam.CFrame.LookVector
		local right = cam.CFrame.RightVector
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + look; horizontalKey = true end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - look; horizontalKey = true end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + right; horizontalKey = true end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - right; horizontalKey = true end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) or flyUp then move = move + Vector3.new(0,1,0) end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) or flyDown then move = move - Vector3.new(0,1,0) end
	elseif flyUp or flyDown then
		if flyUp then move = move + Vector3.new(0,1,0) end
		if flyDown then move = move - Vector3.new(0,1,0) end
	end
	if not horizontalKey then
		local char = LocalPlayer.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if hum and hum.MoveDirection.Magnitude > 0.05 then move = move + hum.MoveDirection end
	end
	return move, horizontalKey
end

local flyObjs = nil

local function flyRelease()
	if flyObjs then
		local o = flyObjs
		flyObjs = nil
		pcall(function() if o.bv then o.bv:Destroy() end end)
		pcall(function() if o.bg then o.bg:Destroy() end end)
	end
end

local function flyEnsure(part)
	if flyObjs and flyObjs.part == part and flyObjs.bv and flyObjs.bv.Parent == part then return flyObjs end
	flyRelease()
	local bv = Instance.new("BodyVelocity")
	bv.Name = "BebrackaFlyVel"; bv.MaxForce = Vector3.new(1e9,1e9,1e9)
	bv.P = 1e5; bv.Velocity = Vector3.new(0,0,0); bv.Parent = part
	local cf = part.CFrame
	local bg = Instance.new("BodyGyro")
	bg.Name = "BebrackaFlyGyro"; bg.MaxTorque = Vector3.new(1e9,1e9,1e9)
	bg.P = 1e4; bg.D = 600; bg.CFrame = cf - cf.Position; bg.Parent = part
	flyObjs = { part = part, bv = bv, bg = bg }
	return flyObjs
end

connect(RunService.Heartbeat, function(dt)
	if not flyOn or spinning then flyRelease(); return end
	local seat, part = getRide()
	if not part or not part.Parent then flyRelease(); return end
	local objs = flyEnsure(part)
	local move = flyInputVector()
	local vel = Vector3.new(0,0,0)
	if move.Magnitude > 0.01 then vel = move.Unit * CONFIG.flySpeed end
	pcall(function()
		objs.bv.Velocity = vel
		local cf = part.CFrame
		local rot = cf - cf.Position
		local targetRot = rot
		local flatMove = Vector3.new(move.X, 0, move.Z)
		local cam = workspace.CurrentCamera
		if cam and flatMove.Magnitude > 0.01 then
			local camLook = cam.CFrame.LookVector
			local camFlat = Vector3.new(camLook.X, 0, camLook.Z)
			local ref = (seat or part).CFrame.LookVector
			local refFlat = Vector3.new(ref.X, 0, ref.Z)
			if camFlat.Magnitude > 0.01 and refFlat.Magnitude > 0.01 then
				local target = math.atan2(-camFlat.X, -camFlat.Z)
				local current = math.atan2(-refFlat.X, -refFlat.Z)
				local diff = (target - current + math.pi) % (2*math.pi) - math.pi
				targetRot = CFrame.Angles(0, diff, 0) * rot
			end
		end
		objs.bg.CFrame = targetRot
	end)
end)

local function visRestoreAll()
	local cache = visCache
	visCache = nil
	if not cache then return end
	for _, info in ipairs(cache.parts) do
		local p = info.part
		if p and p.Parent then
			pcall(function()
				p.Color = info.color; p.Material = info.material
				p.Transparency = info.transp; p.Reflectance = info.refl or 0
			end)
		end
	end
	for key, obj in pairs(cache.objs) do
		pcall(function() obj:Destroy() end)
		cache.objs[key] = nil
	end
end

local function visBuild(model, root)
	local cache = { model = model, root = root, parts = {}, objs = {} }
	local char = LocalPlayer.Character
	local list = model and model:GetDescendants() or { root }
	for i = 1, math.min(#list, 1500) do
		local d = list[i]
		if d:IsA("BasePart") and not (char and d:IsDescendantOf(char)) and d.Transparency < 0.95 then
			table.insert(cache.parts, { part = d, color = d.Color, material = d.Material, transp = d.Transparency, refl = d.Reflectance })
		end
	end
	return cache
end

local function visRainbowSequence(hue)
	local keys = {}
	for i = 0, 6 do
		local tpos = i / 6
		table.insert(keys, ColorSequenceKeypoint.new(tpos, Color3.fromHSV((hue + tpos) % 1, 0.85, 1)))
	end
	return ColorSequence.new(keys)
end

local ctrlHeld, sHeld, wHeld = false, false, false

local function ridePitch()
	local seat, part = getRide()
	if not part or not part.Parent then return nil end
	local ok, y = pcall(function() return part.CFrame.LookVector.Y end)
	if not ok or not y then return nil end
	return math.asin(math.clamp(y, -1, 1))
end

local function rideSpeed()
	local seat, part = getRide()
	if not part or not part.Parent then return 0 end
	local ok, vel = pcall(function() return part.AssemblyLinearVelocity end)
	if ok and vel then return vel.Magnitude end
	ok, vel = pcall(function() return part.Velocity end)
	if ok and vel then return vel.Magnitude end
	return 0
end

local function startSpin()

	if not spinStart() then return false end
	pcall(doJump)

	if spinning then pcall(spinStop); return end
	startSpin()

local function autoSpinLoop()
	if autoSpinRunning then return end
	autoSpinRunning = true
	while autoSpinOn do
		if not gui or not gui.Parent then autoSpinRunning = false; return end
		if autoWheelieOn then task.wait(0.5)
		else
			local cycleStart = os.clock()
			if not spinning then
				if not startSpin() then task.wait(0.5); cycleStart = nil end
			end
			while autoSpinOn and spinning do task.wait(0.05) end
			if autoSpinOn and cycleStart then
				local left = CONFIG.spinInterval - (os.clock() - cycleStart)
				if left > 0 then task.wait(left) end
			end
		end
	end
	autoSpinRunning = false
end

local function autoWheelieLoop()
	if autoWheelieRunning then return end
	autoWheelieRunning = true
	while autoWheelieOn do
		if not gui or not gui.Parent then releaseRideKeys(); autoWheelieRunning = false; return end
		local seat, part = getRide()
		local rideOk = part and part.Parent and not part:IsDescendantOf(LocalPlayer.Character)
		if rideOk and not autoSpinOn then break end
		releaseRideKeys()
		task.wait(0.3)
	end
	if autoWheelieOn then
		holdW(true)
		local tStart = os.clock()
		while autoWheelieOn and gui and gui.Parent and os.clock() - tStart < 3 do
			local seat, part = getRide()
			local rideOk = part and part.Parent and not part:IsDescendantOf(LocalPlayer.Character)
			if not rideOk then break end
			if rideSpeed() > 10 then break end
			task.wait(0.1)
		end
		if not autoWheelieOn or not gui or not gui.Parent then releaseRideKeys(); autoWheelieRunning = false; return end
		local sUp = nil
		local seat, part = getRide()
		local rideOk = part and part.Parent and not part:IsDescendantOf(LocalPlayer.Character)
		if rideOk then
			holdCtrl(true); holdS(true); task.wait(0.3)
			if not autoWheelieOn or not gui or not gui.Parent then releaseRideKeys(); autoWheelieRunning = false; return end
			local p0 = ridePitch()
			holdS(false); task.wait(0.15)
			if not autoWheelieOn or not gui or not gui.Parent then releaseRideKeys(); autoWheelieRunning = false; return end
			local p1 = ridePitch()
			if p0 and p1 then
				if p1 > p0 + math.rad(3) then sUp = false
				elseif p1 < p0 - math.rad(3) then sUp = true end
			end
			if sUp == nil then sUp = false end
			if sUp == false then holdS(true) end
		else releaseRideKeys() end
		local target = math.rad(WHEELIE_ANGLE)
		while autoWheelieOn do
			if not gui or not gui.Parent then releaseRideKeys(); autoWheelieRunning = false; return end
			local seat2, part2 = getRide()
			local rideOk2 = part2 and part2.Parent and not part2:IsDescendantOf(LocalPlayer.Character)
			if not rideOk2 or autoSpinOn then releaseRideKeys(); task.wait(0.3)
			else
				holdW(true); holdCtrl(true)
				local pitch = ridePitch()
				if pitch then
					if sUp == true then holdS(pitch < target and (os.clock() % 0.9) < 0.4)
					elseif sUp == false then holdS(pitch > target + math.rad(8))
					else holdS(false) end
				end
				task.wait(0.05)
			end
		end
	end
	releaseRideKeys()
	autoWheelieRunning = false
end

toggleAutoSpin = function()
	autoSpinOn = not autoSpinOn
	if autoSpinOn then
		if autoWheelieOn then toggleWheelie() end
		task.spawn(autoSpinLoop)
	else if spinning then pcall(spinStop) end end
	if updateUI then pcall(updateUI) end
end

toggleWheelie = function()
	autoWheelieOn = not autoWheelieOn
	if autoWheelieOn then
		if autoSpinOn then toggleAutoSpin() end
		task.spawn(autoWheelieLoop)
	else releaseRideKeys() end
	if updateUI then pcall(updateUI) end
end

	local data = {}
	for i = 0, count - 1 do
		local tm = i / sr
		local v = 0
		for _, nt in ipairs(notes) do
			local x = tm - nt[2]
			if x >= 0 and x < nt[3] then
				local env = (1 - math.exp(-x * 400)) * math.exp(-x * (5 / nt[3]))
				local fade = math.min(1, (nt[3] - x) / 0.02)
				local w = math.sin(2 * math.pi * nt[1] * x) + 0.3 * math.sin(4 * math.pi * nt[1] * x) * math.exp(-x * 20)
				v = v + w * env * fade * nt[4]
			end
		end
		v = math.clamp(v, -1, 1)
		local n = math.floor(v * 30000)
		if n < 0 then n = n + 65536 end
		data[#data + 1] = string.char(n % 256, math.floor(n / 256))
	end
	local pcm = table.concat(data)
	return "RIFF" .. le(36 + #pcm, 4) .. "WAVE" .. "fmt " .. le(16, 4) .. le(1, 2) .. le(1, 2) .. le(sr, 4) .. le(sr * 2, 4) .. le(2, 2) .. le(16, 2) .. "data" .. le(#pcm, 4) .. pcm
end
local function soundFallback(snd, name)
	snd.SoundId = "rbxasset://sounds/switch.wav"
	snd.PlaybackSpeed = (name == "on" and 1.2) or (name == "off" and 0.8) or 1.5
end
function initSounds(parentObj)
	local getAsset = nil
	pcall(function() getAsset = getcustomasset or getsynasset end)
	local canWrite = type(getAsset) == "function" and type(writefile) == "function"
	for name, notes in pairs(SOUND_DEFS) do
		local snd = Instance.new("Sound")
		snd.Name = "Bebracka_" .. name
		snd.Volume = SOUND_VOLUME
		local ok = false
		if canWrite then ok = pcall(function()
			local file = "bebracka_" .. name .. ".wav"
			writefile(file, makeWav(notes))
			snd.SoundId = getAsset(file)
		end) end
		if not ok or snd.SoundId == "" then soundFallback(snd, name) end
		snd.Parent = parentObj
		sounds[name] = snd
		if ok then
			task.spawn(function()
				local t0 = os.clock()
				while snd.Parent and not snd.IsLoaded and os.clock() - t0 < 2.5 do task.wait(0.1) end
				if snd.Parent and not snd.IsLoaded then soundFallback(snd, name) end
			end)
		end
	end
end
function playSound(name)
	local snd = sounds[name]
	if snd and snd.Parent then pcall(function() snd.TimePosition = 0; snd:Play() end) end
end
end

	table.insert(langItems, { obj = label, get = function() return t(titleKey) .. ": " .. currentValue end })
	label.Text = t(titleKey) .. ": " .. currentValue
	local function setFromX(x)
		local absX = track.AbsolutePosition.X
		local width = track.AbsoluteSize.X
		if width <= 0 then return end
		local alpha = math.clamp((x - absX) / width, 0, 1)
		local value = math.floor(minValue + (maxValue - minValue) * alpha + 0.5)
		currentValue = value
		local a = (value - minValue) / (maxValue - minValue)
		fill.Size = UDim2.new(a,0,1,0)
		knob.Position = UDim2.new(a,0,0.5,0)
		label.Text = t(titleKey) .. ": " .. value
		onChange(value)
	end

		BackgroundTransparency = 1, Text = "",

		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliding = true; setFromX(input.Position.X)
		end
	end)
	connect(UserInputService.InputChanged, function(input)
		if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			setFromX(input.Position.X)
		end
	end)
	connect(UserInputService.InputEnded, function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliding = false
		end
	end)
end

connect(UserInputService.InputBegan, function(input, gameProcessed)
	if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
	if listening and input.KeyCode == Enum.KeyCode.Escape then
		listening = nil; refreshBindButtons(); return
	end
	if gameProcessed then return end
	if listening then
		local action = listening
		listening = nil
		if input.KeyCode == Enum.KeyCode.Backspace then binds[action] = nil
		else binds[action] = input.KeyCode end
		refreshBindButtons()
		return
	end
	if autoWheelieOn and (input.KeyCode == Enum.KeyCode.LeftControl or input.KeyCode == Enum.KeyCode.W or input.KeyCode == Enum.KeyCode.S) then return end
	for action, key in pairs(binds) do
		if key and key == input.KeyCode then
			if action == "jump" then pcall(doJump)
			elseif action == "spin" then toggleSpin()
			elseif action == "autospin" then toggleAutoSpin(); playSound(autoSpinOn and "on" or "off")
			elseif action == "wheelie" then toggleWheelie(); playSound(autoWheelieOn and "on" or "off")
			elseif action == "speed" then toggleSpeed(); playSound(speedOn and "on" or "off")
			elseif action == "fly" then toggleFly(); playSound(flyOn and "on" or "off") end
			break
		end
	end
end)

end
