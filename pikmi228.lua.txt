-- Функция для ручного перетаскивания (более плавное)
local dragging = false
local dragInput = nil
local dragStart = nil
local startPos = nil

local function updateDrag(input)
    local delta = input.Position - dragStart
    SecretFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

SecretTitle.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = SecretFrame.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

SecretTitle.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and input == dragInput then
        updateDrag(input)
    end
end)

-- Обработка нажатия L
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.L then
        SecretScreen.Enabled = not SecretScreen.Enabled
    end
end)

-- Система ключей
local validKeys = {
    "pikmi228",
    "sandra"
}

local function checkKey(key)
    for _, validKey in ipairs(validKeys) do
        if key == validKey then
            return true
        end
    end
    return false
end

local keyAccepted = false

local function activateKey()
    local enteredKey = KeyBox.Text
    if checkKey(enteredKey) then
        keyAccepted = true
        KeyStatus.Text = "✅ Ключ верный!"
        KeyStatus.TextColor3 = Color3.fromRGB(0, 255, 0)
        wait(1)
        KeyScreen:Destroy()
        SecretScreen:Destroy()
        loadMainGUI()
    else
        KeyStatus.Text = "❌ Неверный ключ!"
        KeyStatus.TextColor3 = Color3.fromRGB(255, 0, 0)
        KeyBox.Text = ""
    end
end

KeyButton.MouseButton1Click:Connect(activateKey)

KeyBox.FocusLost:Connect(function(enterPressed)
    if enterPressed then
        activateKey()
    end
end)
