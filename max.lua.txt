    if RaycastResult and RaycastResult.Instance and RaycastResult.Instance:IsDescendantOf(MurdererCharacter) then
        pcall(function()
            GunTool:Activate()
        end)
    end
end

-- WALL HOP
local WallHopConfig = {Enabled = false, HopPower = 60, ForwardCheck = 3, RotationBounce = true, RotationDelay = 0.12}
local wallHopCharacter = Player.Character or Player.CharacterAdded:Wait()
local function toggleWallHop(v)
    WallHopConfig.Enabled = v
end
local function getWallHopComponents()
    if not wallHopCharacter or not wallHopCharacter.Parent then wallHopCharacter = Player.Character or Player.CharacterAdded:Wait() end
    return wallHopCharacter, wallHopCharacter:FindFirstChildOfClass("Humanoid"), wallHopCharacter:FindFirstChild("HumanoidRootPart")
end
local function isTouchingWall(hrp)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {wallHopCharacter}
    params.FilterType = Enum.RaycastFilterType.Blacklist
    return workspace:Raycast(hrp.Position, hrp.CFrame.LookVector * WallHopConfig.ForwardCheck, params)
end
local function performHop()
    local char, humanoid, hrp = getWallHopComponents()
    if not (humanoid and hrp) or humanoid:GetState() == Enum.HumanoidStateType.Dead then return end
    if humanoid.FloorMaterial == Enum.Material.Air and isTouchingWall(hrp) then
        humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
        hrp.Velocity = Vector3.new(hrp.Velocity.X, WallHopConfig.HopPower, hrp.Velocity.Z)
        if WallHopConfig.RotationBounce then
            local originalAutoRotate = humanoid.AutoRotate
            humanoid.AutoRotate = false
            hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(-180), 0)
            task.delay(WallHopConfig.RotationDelay, function()
                if hrp and hrp.Parent then hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(180), 0) end
                if humanoid and humanoid.Parent then humanoid.AutoRotate = originalAutoRotate end
            end)
        end
    end
end
UserInputService.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if WallHopConfig.Enabled and input.KeyCode == Enum.KeyCode.E then performHop() end
end)

-- ROLES
local function getRole(plr)
    if not plr or not plr.Character then return "Innocent" end
    local char = plr.Character
    for _, tool in ipairs(char:GetChildren()) do if tool:IsA("Tool") then
        if tool.Name:lower():find("knife") then return "Murderer" end
        if tool.Name:lower():find("gun") then return "Sheriff" end
    end end
    if plr.Backpack then for _, tool in ipairs(plr.Backpack:GetChildren()) do if tool:IsA("Tool") then
        if tool.Name:lower():find("knife") then return "Murderer" end
        if tool.Name:lower():find("gun") then return "Sheriff" end
    end end end
    return "Innocent"
end
local function getRoleColor(role)
    if role == "Murderer" then return Color3.fromRGB(220, 40, 40) end
    if role == "Sheriff" then return Color3.fromRGB(40, 120, 255) end
    return Color3.fromRGB(50, 205, 50)
end

    end
end

-- ESP
local DEFAULT_COLOR = Color3.fromRGB(50, 205, 50)
local KNIFE_COLOR = Color3.fromRGB(220, 40, 40)
local GUN_COLOR = Color3.fromRGB(40, 120, 255)
local HL_TAG = "_WH"
local BB_TAG = "_WBB"
local trackedData = {}
local espEnabled = true
local weaponKeywords = {
    { pattern = "gun", priority = 2, color = GUN_COLOR },
    { pattern = "knife", priority = 1, color = KNIFE_COLOR },
}
local function clearESP(player, character)
    local hl = character:FindFirstChild(HL_TAG)
    if hl then hl:Destroy() end
    local head = character:FindFirstChild("Head")
    if head then
        local bb = head:FindFirstChild(BB_TAG)
        if bb then bb:Destroy() end
    end
end
local function detectWeaponColor(character, player)
    local best, bestPrio = nil, 0
    local function checkTool(tool)
        if not tool:IsA("Tool") then return end
        local lower = tool.Name:lower()
        for _, entry in ipairs(weaponKeywords) do
            if lower:find(entry.pattern) and entry.priority > bestPrio then
                bestPrio = entry.priority
                best = entry.color
            end
        end
    end
    for _, obj in ipairs(character:GetChildren()) do checkTool(obj) end
    local bp = player:FindFirstChildOfClass("Backpack")
    if bp then for _, obj in ipairs(bp:GetChildren()) do checkTool(obj) end end
    return best or DEFAULT_COLOR
end
local function getOrMakeHighlight(character)
    local h = character:FindFirstChild(HL_TAG)
    if h and h:IsA("Highlight") then return h end
    if h then h:Destroy() end
    local hl = Instance.new("Highlight")
    hl.Name = HL_TAG
    hl.Adornee = character
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.FillTransparency = 1
    hl.OutlineTransparency = 0
    hl.FillColor = DEFAULT_COLOR
    hl.OutlineColor = DEFAULT_COLOR
    hl.Parent = character
    return hl
end
local function getOrMakeBillboard(character, playerName)
    local head = character:FindFirstChild("Head")
    if not head then return nil end
    local bb = head:FindFirstChild(BB_TAG)
    if bb and bb:IsA("BillboardGui") then return bb:FindFirstChild("NameLabel") end
    if bb then bb:Destroy() end
    local billboard = Instance.new("BillboardGui")
    billboard.Name = BB_TAG
    billboard.Adornee = head
    billboard.AlwaysOnTop = true
    billboard.Size = UDim2.new(0, 140, 0, 26)
    billboard.StudsOffset = Vector3.new(0, 2.6, 0)
    billboard.ResetOnSpawn = false
    billboard.Parent = head
    local label = Instance.new("TextLabel")
    label.Name = "NameLabel"
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = playerName
    label.Font = Enum.Font.GothamBold
    label.TextSize = 13
    label.TextColor3 = DEFAULT_COLOR
    label.TextStrokeTransparency = 1
    label.Parent = billboard
    return label
end
local function updatePlayer(player)
    if player == Player then return end
    local character = player.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    if not espEnabled then clearESP(player, character) return end
    local color = detectWeaponColor(character, player)
    local hl = getOrMakeHighlight(character)
    hl.OutlineColor = color
    hl.FillColor = color
    local label = getOrMakeBillboard(character, player.DisplayName)
    if label then label.TextColor3 = color end
end
local function hookCharacter(player, character)
    character:WaitForChild("HumanoidRootPart", 10)
    local conn = RunService.Heartbeat:Connect(function()
        if not character.Parent then return end
        updatePlayer(player)
    end)
    local prev = trackedData[player]
    if prev then
        if prev.conn then prev.conn:Disconnect() end
        if prev.character then clearESP(player, prev.character) end
    end
    trackedData[player] = { conn = conn, character = character }
    character.AncestryChanged:Connect(function(_, parent)
        if not parent and trackedData[player] and trackedData[player].character == character then conn:Disconnect() end
    end)
end
local function hookPlayer(player)
    if player == Player then return end
    if player.Character then task.spawn(hookCharacter, player, player.Character) end
    player.CharacterAdded:Connect(function(character) task.spawn(hookCharacter, player, character) end)
end
local function toggleESP(v)
    espEnabled = v
    if not v then
        for p, data in pairs(trackedData) do
            if data.character then clearESP(p, data.character) end
        end
    end
end
for _, player in ipairs(Players:GetPlayers()) do task.spawn(hookPlayer, player) end
Players.PlayerAdded:Connect(hookPlayer)
Players.PlayerRemoving:Connect(function(player)
    local data = trackedData[player]
    if not data then return end
    if data.conn then data.conn:Disconnect() end
    if data.character then clearESP(player, data.character) end
    trackedData[player] = nil
end)

-- ===== АИМБОТ 1 (СТАРЫЙ) =====
local aimbotEnabled = false
local aimbotConnection = nil
local aimbotTarget = "Head"
local aimbotFOV = 1000
local aimbotSmoothness = 0.5
local function getNearestMurderer()
    local root = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local murderer, dist = nil, aimbotFOV
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player and plr.Character then
            local tr = plr.Character:FindFirstChild("HumanoidRootPart")
            if tr and getRole(plr) == "Murderer" then
                local d = (tr.Position - root.Position).Magnitude
                if d < dist then dist = d murderer = plr end
            end
        end
    end
    return murderer
end
local function toggleAimbot(v)
    aimbotEnabled = v
    if aimbotConnection then aimbotConnection:Disconnect() aimbotConnection = nil end
    if v then
        aimbotConnection = RunService.RenderStepped:Connect(function()
            if not aimbotEnabled then return end
            local char = Player.Character
            if not char then return end
            local root = char:FindFirstChild("HumanoidRootPart")
            if not root then return end
            local m = getNearestMurderer()
            if m and m.Character then
                local targetPart = aimbotTarget == "Head" and m.Character:FindFirstChild("Head") or m.Character:FindFirstChild("HumanoidRootPart")
                if targetPart then
                    local targetPos = targetPart.Position
                    local rootPos = root.Position
                    local flatTarget = Vector3.new(targetPos.X, rootPos.Y, targetPos.Z)
                    if (flatTarget - rootPos).Magnitude > 0.5 then
                        root.CFrame = root.CFrame:Lerp(CFrame.lookAt(rootPos, flatTarget), aimbotSmoothness)
                    end
                    Camera.CFrame = Camera.CFrame:Lerp(CFrame.lookAt(Camera.CFrame.Position, targetPos), aimbotSmoothness)
                end
            end
        end)
    end
end
local function setAimbotTarget(target)
    aimbotTarget = target
    if aimbotEnabled then toggleAimbot(false) toggleAimbot(true) end
end

local function CreateColorPicker(parent, name, yPos, defaultColor, callback)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, -20, 0, 30)
    btn.Position = UDim2.new(0, 10, 0, yPos)
    btn.BackgroundColor3 = defaultColor
    btn.Text = name
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.TextScaled = true
    btn.Font = Enum.Font.Gotham
    btn.Parent = parent
    local colors = {Color3.fromRGB(255,0,0), Color3.fromRGB(0,255,0), Color3.fromRGB(0,0,255), Color3.fromRGB(255,255,0), Color3.fromRGB(255,0,255), Color3.fromRGB(0,255,255), Color3.fromRGB(255,255,255), Color3.fromRGB(255,100,0)}
    local colorIndex = 1
    for i, c in ipairs(colors) do if c == defaultColor then colorIndex = i break end end
    btn.MouseButton1Click:Connect(function()

local function RemoveExtraEffect(name)
    if extraEffects[name] then
        if extraEffects[name]:IsA("ImageLabel") or extraEffects[name]:IsA("Frame") then
            extraEffects[name]:Destroy()
        else
            pcall(function() extraEffects[name]:Destroy() end)
        end
        extraEffects[name] = nil
    end
end

local function ResetAllEffects()
    for i, v in pairs(Lighting:GetChildren()) do
        if v:IsA("BloomEffect") or v:IsA("BlurEffect") or v:IsA("ColorCorrectionEffect") or 
           v:IsA("SunRaysEffect") or v:IsA("Sky") or v:IsA("Atmosphere") then
            v:Destroy()
        end
    end
    Lighting.Ambient = Color3.fromRGB(127, 127, 127)
    Lighting.Brightness = 2
    Lighting.ClockTime = 12
    Lighting.ExposureCompensation = 0
    for name, obj in pairs(extraEffects) do
        pcall(function() obj:Destroy() end)
    end
    extraEffects = {}
    if overlayGui then
        overlayGui:Destroy()
        overlayGui = nil
    end
    effects = {}
end

local function CreateSlider(parent, name, yPos, minVal, maxVal, defaultVal, callback)
    local container = Instance.new("Frame")
    container.Size = UDim2.new(1, -20, 0, 40)
    container.Position = UDim2.new(0, 10, 0, yPos)
    container.BackgroundTransparency = 1
    container.Parent = parent
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 0, 15)
    label.BackgroundTransparency = 1
    label.Text = name .. ": " .. defaultVal
    label.TextColor3 = Color3.fromRGB(255, 255, 255)
    label.TextScaled = true
    label.Font = Enum.Font.Gotham
    label.Parent = container
    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, 0, 0, 10)
    track.Position = UDim2.new(0, 0, 0, 25)
    track.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
    track.Parent = container
    Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)
    local fill = Instance.new("Frame")
    fill.Size = UDim2.new(0, 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(0, 150, 0)
    fill.Parent = track
    local handle = Instance.new("TextButton")
    handle.Size = UDim2.new(0, 18, 0, 18)
    handle.Position = UDim2.new(0, -9, 0.5, -9)
    handle.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    handle.Text = ""
    handle.Parent = track
    Instance.new("UICorner", handle).CornerRadius = UDim.new(1, 0)
    local currentValue = defaultVal
    local dragging = false
    local function setFromPosition(x)
        local trackWidth = track.AbsoluteSize.X
        local relativeX = math.clamp(x, 0, trackWidth)
        local percent = relativeX / trackWidth
        currentValue = minVal + (maxVal - minVal) * percent
        currentValue = math.round(currentValue * 100) / 100
        label.Text = name .. ": " .. currentValue
        handle.Position = UDim2.new(percent, -9, 0.5, -9)
        fill.Size = UDim2.new(percent, 0, 1, 0)
        callback(currentValue)
    end
    handle.MouseButton1Down:Connect(function(input) dragging = true setFromPosition(input.Position.X - track.AbsolutePosition.X) end)
    UserInputService.InputEnded:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            setFromPosition(input.Position.X - track.AbsolutePosition.X)
        end
    end)
    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = true setFromPosition(input.Position.X - track.AbsolutePosition.X) end
    end)
    setFromPosition((defaultVal - minVal) / (maxVal - minVal) * track.AbsoluteSize.X)
    return { GetValue = function() return currentValue end }
end
